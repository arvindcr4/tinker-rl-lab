```sql
-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- 1. Authentication (Handled by Supabase Auth, but we need a profile table)
create table public.profiles (
  id uuid references auth.users not null primary key,
  email text,
  username text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.profiles enable row level security;

create policy "Public profiles are viewable by everyone."
  on public.profiles for select
  using ( true );

create policy "Users can insert their own profile."
  on public.profiles for insert
  with check ( auth.uid() = id );

create policy "Users can update own profile."
  on public.profiles for update
  using ( auth.uid() = id );

-- Trigger to create profile on signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, email, username)
  values (new.id, new.email, split_part(new.email, '@', 1));
  return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- 2. Public Chat Room
create table public.chat_messages (
  id uuid default uuid_generate_v4() primary key,
  user_id uuid references auth.users not null,
  message text not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.chat_messages enable row level security;

create policy "Messages are viewable by authenticated users."
  on public.chat_messages for select
  using ( auth.role() = 'authenticated' );

create policy "Authenticated users can insert messages."
  on public.chat_messages for insert
  with check ( auth.role() = 'authenticated' );

-- 3. AI Chat History (Private per user)
create table public.ai_chats (
  id uuid default uuid_generate_v4() primary key,
  user_id uuid references auth.users not null,
  role text not null, -- 'user' or 'assistant'
  content text not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.ai_chats enable row level security;

create policy "Users can view own AI chats."
  on public.ai_chats for select
  using ( auth.uid() = user_id );

create policy "Users can insert own AI chats."
  on public.ai_chats for insert
  with check ( auth.uid() = user_id );
```

const openai = new OpenAI({

export async function generateAIResponse(
  messages: { role: string; content: string }[],
  context?: string
) {
  const systemPrompt = `You are a financial analyst assistant integrated into a Bloomberg Terminal. 
  You provide concise, data-driven insights. 
  If a date range is provided in the context, you must explicitly reference it in your analysis.
  If the user asks for news or current events, simulate a web search by providing the most relevant recent financial context you have, or general market trends if specific real-time data isn't available in the prompt.`;

  const fullMessages = [
    { role: 'system', content: systemPrompt },
    ...(context ? [{ role: 'system', content: `CURRENT CONTEXT: ${context}` }] : []),
    ...messages
  ];

  try {
    const response = await openai.chat.completions.create({
      model: "gpt-3.5-turbo",
      messages: fullMessages,
      temperature: 0.7,
    });
    return response.choices[0].message.content;
  } catch (error) {
    console.error("OpenAI Error:", error);
    return "Error generating response. Please try again.";
  }
}

'use client';

import { createContext, useContext, useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import type { Session, User } from '@supabase/supabase-js';

type AuthContextType = {
  session: Session | null;
  user: User | null;
  loading: boolean;
  signOut: () => Promise<void>;
};

const AuthContext = createContext<AuthContextType>({} as AuthContextType);

export const AuthProvider = ({ children }: { children: React.ReactNode }) => {
  const [session, setSession] = useState<Session | null>(null);
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Set session from local storage immediately
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    return () => subscription.unsubscribe();
  }, []);

  const signOut = async () => {
    await supabase.auth.signOut();
  };

  return (
    <AuthContext.Provider value={{ session, user, loading, signOut }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useSession = () => useContext(AuthContext);

export default function Login() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [isSignUp, setIsSignUp] = useState(false);
  const [error, setError] = useState('');

    setError('');

      else setError('Check your email for confirmation link.');

      if (error) setError(error.message);

'use client';

import React, { useEffect, useState, useRef } from 'react';
import { 
  LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, 
  ReferenceArea, Brush 
} from 'recharts';

interface StockChartProps {
  symbol: string;
  onSelectionChange: (start: Date, end: Date) => void;
}

const StockChart: React.FC<StockChartProps> = ({ symbol, onSelectionChange }) => {
  const [data, setData] = useState<any[]>([]);
  const [selectedRange, setSelectedRange] = useState<{x1: number, x2: number} | null>(null);
  const chartRef = useRef<any>(null);

  // Generate mock stock data
  useEffect(() => {
    const generateData = () => {
      const now = new Date();
      const dataPoints = [];
      let price = 150 + Math.random() * 50;
      
      for (let i = 0; i < 100; i++) {
        const time = new Date(now.getTime() - (100 - i) * 60000); // Last 100 minutes
        const change = (Math.random() - 0.5) * 2;
        price += change;
        dataPoints.push({
          time: time.toISOString(),
          price: parseFloat(price.toFixed(2)),
          originalTime: time
        });
      }
      return dataPoints;
    };

    setData(generateData());

    // Simulate real-time updates
    const interval = setInterval(() => {
      setData(prev => {
        const last = prev[prev.length - 1];
        const newTime = new Date();
        const change = (Math.random() - 0.5) * 2;
        const newPrice = last.price + change;
        
        const newData = [...prev.slice(1), {
          time: newTime.toISOString(),
          price: parseFloat(newPrice.toFixed(2)),
          originalTime: newTime
        }];
        return newData;
      });
    }, 2000);

    return () => clearInterval(interval);
  }, [symbol]);

  const handleMouseDown = (e: any) => {
    if (e.activeLabelIndex !== undefined) {
      setSelectedRange({ x1: e.activeLabelIndex, x2: e.activeLabelIndex });
    }
  };

  const handleMouseMove = (e: any) => {
    if (selectedRange && e.activeLabelIndex !== undefined) {
      setSelectedRange({ 
        x1: Math.min(selectedRange.x1, e.activeLabelIndex), 
        x2: Math.max(selectedRange.x1, e.activeLabelIndex) 
      });
    }
  };

  const handleMouseUp = () => {
    if (selectedRange && selectedRange.x1 !== selectedRange.x2 && chartRef.current) {
      const xScale = chartRef.current.internal.xScale;
      const startIdx = Math.min(selectedRange.x1, selectedRange.x2);
      const endIdx = Math.max(selectedRange.x1, selectedRange.x2);
      
      const startDate = data[startIdx]?.originalTime;
      const endDate = data[endIdx]?.originalTime;

      if (startDate && endDate) {
        onSelectionChange(startDate, endDate);
      }
    }
    setSelectedRange(null);
  };

  return (
    <div className="w-full h-full bg-terminal-panel border border-terminal-border p-2">
      <ResponsiveContainer width="100%" height="100%">
        <LineChart
          data={data}
          onMouseDown={handleMouseDown}
          onMouseMove={handleMouseMove}
          onMouseUp={handleMouseUp}
          ref={chartRef}
        >
          <CartesianGrid strokeDasharray="3 3" stroke="#333" />
          <XAxis 
            dataKey="time" 
            tickFormatter={(val) => new Date(val).toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})} 
            stroke="#666"
            tick={{fontSize: 10}}
          />
          <YAxis 
            domain={['auto', 'auto']} 
            stroke="#666"
            tick={{fontSize: 10}}
          />
          <Tooltip 
            contentStyle={{ backgroundColor: '#111', border: '1px solid #333', color: '#fff' }}
            labelFormatter={(val) => new Date(val).toLocaleString()}
          />
          <Line 
            type="monotone" 
            dataKey="price" 
            stroke="#00ff00" 
            dot={false} 
            activeDot={{ r: 4 }}
          />
          {selectedRange && (
            <ReferenceArea 
              x1={selectedRange.x1} 
              x2={selectedRange.x2} 
              fill="#00ff00" 
              fillOpacity={0.1} 
              strokeOpacity={0}
            />
          )}
          <Brush 
            dataKey="time"

'use client';

import { useState, useEffect, useRef } from 'react';
import { supabase } from '@/lib/supabase';
import { useSession } from './AuthProvider';
import { generateAIResponse } from '@/lib/openai';

interface AIChatProps {
  context?: string;
}

export default function AIChat({ context }: AIChatProps) {
  const { user } = useSession();
  const [messages, setMessages] = useState<{role: string, content: string}[]>([]);
  const [input, setInput] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const messagesEndRef = useRef<HTMLDivElement>(null);

  // Load history
  useEffect(() => {
    if (!user) return;

    const loadHistory = async () => {
      const { data, error } = await supabase
        .from('ai_chats')
        .select('*')
        .eq('user_id', user.id)
        .order('created_at', { ascending: true });
      
      if (data) {
        setMessages(data.map((m: any) => ({ role: m.role, content: m.content })));
      }
    };

    loadHistory();

    // Listen for new messages (if shared, though this is private)
    const channel = supabase
      .channel('ai_chats_changes')
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'ai_chats', filter: `user_id=eq.${user.id}` }, (payload) => {
        setMessages(prev => [...prev, { role: payload.new.role, content: payload.new.content }]);
      })
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [user]);

  // Auto-scroll
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSend = async () => {
    if (!input.trim() || !user) return;

    const userMsg = { role: 'user', content: input };
    const newMessages = [...messages, userMsg];
    
    setMessages(newMessages);
    setInput('');
    setIsLoading(true);

    // Save user message
    await supabase.from('ai_chats').insert({ user_id: user.id, role: 'user', content: input });

    // Get AI Response
    const response = await generateAIResponse(newMessages, context);
    
    const aiMsg = { role: 'assistant', content: response };
    setMessages(prev => [...prev, aiMsg]);
    
    // Save AI message
    await supabase.from('ai_chats').insert({ user_id: user.id, role: 'assistant', content: response });
    setIsLoading(false);
  };

  return (
    <div className="flex-1 flex flex-col overflow-hidden bg-terminal-bg">
      <div className="flex-1 overflow-y-auto p-2 space-y-2">
        {messages.map((msg, idx) => (
          <div key={idx} className={`text-xs ${msg.role === 'user' ? 'text-terminal-blue' : 'text-terminal-accent'}`}>
            <span className="font-bold mr-2">{msg.role === 'user' ? 'YOU' : 'AI'}:</span>
            {msg.content}
          </div>
        ))}
        {isLoading && <div className="text-xs text-terminal-text/50">Thinking...</div>}
        <div ref={messagesEndRef} />
      </div>
      <div className="p-2 border-t border-terminal-border flex">
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === 'Enter' && handleSend()}
          placeholder="Ask about the market..."
          className="flex-1 bg-terminal-panel border border-terminal-border text-terminal-text p-2 text-xs focus:outline-none focus:border-terminal-accent"
        />
        <button 
          onClick={handleSend}
          disabled={isLoading}
          className="ml-2 bg-terminal-accent text-terminal-bg px-3 py-2 text-xs font-bold hover:bg-green-400 disabled:opacity-50"
        >
          SEND
        </button>
      </div>
    </div>
  );
}

'use client';

import { useState, useEffect, useRef } from 'react';
import { supabase } from '@/lib/supabase';
import { useSession } from './AuthProvider';

export default function PublicChat() {
  const { user } = useSession();
  const [messages, setMessages] = useState<any[]>([]);
  const [input, setInput] = useState('');
  const messagesEndRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!user) return;

    const loadMessages = async () => {
      const { data } = await supabase
        .from('chat_messages')
        .select('*')
        .order('created_at', { ascending: true })
        .limit(50);
      
      if (data) setMessages(data);
    };

    loadMessages();

    const channel = supabase
      .channel('public_chat_changes')
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'chat_messages' }, (payload) => {
        setMessages(prev => [...prev, payload.new]);
      })
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [user]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSend = async () => {
    if (!input.trim() || !user) return;
    
    await supabase.from('chat_messages').insert({
      user_id: user.id,
      message: input
    });
    
    setInput('');
  };

  return (
    <div className="flex-1 flex flex-col overflow-hidden bg-terminal-bg">
      <div className="flex-1 overflow-y-auto p-2 space-y-2">
        {messages.map((msg) => (
          <div key={msg.id} className="text-xs border-b border-terminal-border/20 pb-1">
            <span className="text-terminal-blue font-bold">[{new Date(msg.created_at).toLocaleTimeString()}]</span>
            <span className="text-terminal-yellow ml-2">{msg.user_id === user?.id ? 'YOU' : 'USER'}</span>
            <span className="text-terminal-text ml-2">{msg.message}</span>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

        <input
          type="text"

          className="flex-1 bg-terminal-panel border border-terminal-border text-terminal-text p-2 text-xs focus:outline-none focus:border-terminal-accent"
        />
        <button 
          onClick={handleSend}
          className="ml-2 bg-terminal-blue text-terminal-bg px-3 py-2 text-xs font-bold hover:bg-blue-400"

          POST
        </button>
      </div>
    </div>
  );
}
