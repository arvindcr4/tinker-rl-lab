Here is the complete implementation for the Hospital Patient Tracking Board.

### Technical Architecture
1.  **Framework**: Next.js 14+ (App Router) with TypeScript.
2.  **Database & Auth**: Supabase (PostgreSQL, Auth, Realtime).
3.  **State Management**: React Context for Auth/Presence, Supabase JS Client for Data/Realtime.
4.  **Styling**: Tailwind CSS.
5.  **Icons**: Lucide React.

### Prerequisites
You must have a Supabase project.
1.  Create a project.
2.  Run the SQL schema provided in `supabase-schema.sql` in the SQL Editor.
3.  Get your `NEXT_PUBLIC_SUPABASE_URL` and `NEXT_PUBLIC_SUPABASE_ANON_KEY`.

'use client'

import { createContext, useContext, useEffect, useState, ReactNode } from 'react'
import { supabase } from '@/lib/supabase'
import { User, Session } from '@supabase/supabase-js'
import { getProfile } from '@/lib/utils'

interface AuthContextType {
  user: User | null
  session: Session | null
  profile: any
  loading: boolean
  signOut: () => Promise<void>
}

const AuthContext = createContext<AuthContextType | undefined>(undefined)

export const AuthProvider = ({ children }: { children: ReactNode }) => {
  const [user, setUser] = useState<User | null>(null)
  const [session, setSession] = useState<Session | null>(null)
  const [profile, setProfile] = useState<any>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    // Get initial session
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session)
      setUser(session?.user ?? null)
      if (session?.user) {
        getProfile(session.user.id).then(setProfile)
      }
      setLoading(false)
    })

    // Listen for changes
    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session)
      setUser(session?.user ?? null)
      if (session?.user) {
        getProfile(session.user.id).then(setProfile)
      } else {
        setProfile(null)
      }
      setLoading(false)
    })

    return () => subscription.unsubscribe()
  }, [])

  const signOut = async () => {
    await supabase.auth.signOut()
  }

  return (
    <AuthContext.Provider value={{ user, session, profile, loading, signOut }}>
      {children}
    </AuthContext.Provider>
  )
}

export const useAuth = () => {
  const context = useContext(AuthContext)
  if (context === undefined) {
    throw new Error('useAuth must be used within an AuthProvider')
  }
  return context
}

'use client'

import { createContext, useContext, useEffect, useState, ReactNode } from 'react'
import { supabase } from '@/lib/supabase'
import { useAuth } from './AuthContext'
import { updateProfileLastSeen } from '@/lib/utils'

interface PresenceContextType {
  onlineUsers: string[]
}

const PresenceContext = createContext<PresenceContextType | undefined>(undefined)

export const PresenceProvider = ({ children }: { children: ReactNode }) => {
  const { user } = useAuth()
  const [onlineUsers, setOnlineUsers] = useState<string[]>([])

  useEffect(() => {
    if (!user) return

    // Update last seen on mount and interval
    const updatePresence = async () => {
      await updateProfileLastSeen(user.id)
    }

    updatePresence()
    const interval = setInterval(updatePresence, 30000) // Every 30s

    // Listen for profile updates (last_seen changes)
    const channel = supabase
      .channel('presence_changes')
      .on(
        'postgres_changes',
        {
          event: 'UPDATE',
          schema: 'public',
          table: 'profiles',
        },
        (payload) => {
          // If the user is still active (last_seen within 1 min), consider them online
          const lastSeen = new Date(payload.new.last_seen)
          const now = new Date()
          if (now.getTime() - lastSeen.getTime() < 60000) {
            setOnlineUsers((prev) => {
              if (!prev.includes(payload.new.id)) {
                return [...prev, payload.new.id]
              }
              return prev
            })
          } else {
            setOnlineUsers((prev) => prev.filter((id) => id !== payload.new.id))
          }
        }
      )
      .subscribe()

    // Initial load of online users
    supabase
      .from('profiles')
      .select('id')
      .eq('is_active', true)
      .then(({ data }) => {
        if (data) {
          setOnlineUsers(data.map((p) => p.id))
        }
      })

    return () => {
      clearInterval(interval)
      channel.unsubscribe()
    }
  }, [user])

  return (
    <PresenceContext.Provider value={{ onlineUsers }}>
      {children}
    </PresenceContext.Provider>
  )
}

export const usePresence = () => {
  const context = useContext(PresenceContext)
  if (context === undefined) {
    throw new Error('usePresence must be used within a PresenceProvider')
  }
  return context
}

'use client'

import { createContext, useContext, useEffect, useState, ReactNode } from 'react'
import { supabase } from '@/lib/supabase'
import { useAuth } from './AuthContext'

interface Alert {
  id: string
  title: string
  message: string
  severity: string
  category: string
  is_active: boolean
  created_by: string
  created_at: string
}

interface AlertContextType {
  alerts: Alert[]
  dismissedAlerts: string[]
  addAlert: (alert: Omit<Alert, 'id' | 'created_at' | 'is_active' | 'created_by'>) => Promise<void>
  dismissAlert: (id: string) => void
}

const AlertContext = createContext<AlertContextType | undefined>(undefined)

export const AlertProvider = ({ children }: { children: ReactNode }) => {
  const { user } = useAuth()
  const [alerts, setAlerts] = useState<Alert[]>([])
  const [dismissedAlerts, setDismissedAlerts] = useState<string[]>([])

  useEffect(() => {
    if (!user) return

    // Load active alerts
    const loadAlerts = async () => {
      const { data } = await supabase
        .from('emergency_alerts')
        .select('*')
        .eq('is_active', true)
        .order('created_at', { ascending: false })
      
      if (data) setAlerts(data)
    }

    loadAlerts()

    // Listen for new alerts
    const channel = supabase
      .channel('alerts_channel')
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'emergency_alerts',
        },
        (payload) => {
          if (payload.eventType === 'INSERT') {
            setAlerts((prev) => [payload.new as Alert, ...prev])
          } else if (payload.eventType === 'DELETE') {
            setAlerts((prev) => prev.filter((a) => (a as any).id !== (payload.old as any).id))
          }
        }
      )
      .subscribe()

    return () => {
      channel.unsubscribe()
    }
  }, [user])

  const addAlert = async (alertData: Omit<Alert, 'id' | 'created_at' | 'is_active' | 'created_by'>) => {
    if (!user) return
    await supabase.from('emergency_alerts').insert({
      ...alertData,
      is_active: true,
      created_by: user.id,
    })
  }

  const dismissAlert = (id: string) => {
    setDismissedAlerts((prev) => [...prev, id])
  }

  return (
    <AlertContext.Provider value={{ alerts, dismissedAlerts, addAlert, dismissAlert }}>
      {children}
    </AlertContext.Provider>
  )
}

export const useAlerts = () => {
  const context = useContext(AlertContext)
  if (context === undefined) {
    throw new Error('useAlerts must be used within an AlertProvider')
  }
  return context
}

'use client'

import { createContext, useContext, useEffect, useState, ReactNode } from 'react'
import { supabase } from '@/lib/supabase'
import { useAuth } from './AuthContext'

interface Unit {
  id: string
  name: string
  description: string
}

interface Bed {
  id: string
  unit_id: string
  bed_number: string
  is_active: boolean
}

interface Encounter {
  id: string
  bed_id: string
  patient_name: string
  mrn: string
  dob: string
  status: string
  notes: string
  assigned_nurse_id: string
}

interface Task {
  id: string
  encounter_id: string
  description: string
  is_completed: boolean
}

interface BoardData {
  units: Unit[]
  beds: Bed[]
  encounters: Encounter[]
  tasks: Task[]
}

interface BoardContextType {
  data: BoardData
  loading: boolean
  addUnit: (name: string, description: string) => Promise<void>
  updateUnit: (id: string, name: string, description: string) => Promise<void>
  deleteUnit: (id: string) => Promise<void>
  addBed: (unitId: string, bedNumber: string) => Promise<void>
  updateBed: (id: string, bedNumber: string) => Promise<void>
  deleteBed: (id: string) => Promise<void>
  updateEncounter: (id: string, data: Partial<Encounter>) => Promise<void>
  addTask: (encounterId: string, description: string) => Promise<void>
  updateTask: (id: string, isCompleted: boolean) => Promise<void>
  deleteTask: (id: string) => Promise<void>
}

const BoardContext = createContext<BoardContextType | undefined>(undefined)

export const BoardProvider = ({ children }: { children: ReactNode }) => {
  const { user } = useAuth()
  const [data, setData] = useState<BoardData>({
    units: [],
    beds: [],
    encounters: [],
    tasks: [],
  })
  const [loading, setLoading] = useState(true)

  const loadData = async () => {
    const [unitsRes, bedsRes, encountersRes, tasksRes] = await Promise.all([
      supabase.from('units').select('*').order('name'),
      supabase.from('beds').select('*').order('bed_number'),
      supabase.from('encounters').select('*').order('updated_at', { ascending: false }),
      supabase.from('tasks').select('*').order('created_at', { ascending: false }),
    ])

    if (unitsRes.data && bedsRes.data && encountersRes.data && tasksRes.data) {
      setData({
        units: unitsRes.data,
        beds: bedsRes.data,
        encounters: encountersRes.data,
        tasks: tasksRes.data,
      })
    }
    setLoading(false)
  }

  useEffect(() => {
    if (!user) return
    loadData()

    // Listen for changes
    const channel = supabase
      .channel('board_channel')
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'units',
        },
        () => loadData()
      )
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'beds',
        },
        () => loadData()
      )
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'encounters',
        },
        () => loadData()
      )
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'tasks',
        },
        () => loadData()
      )
      .subscribe()

    return () => {
      channel.unsubscribe()
    }
  }, [user])

  const addUnit = async (name: string, description: string) => {
    await supabase.from('units').insert({ name, description })
  }

  const updateUnit = async (id: string, name: string, description: string) => {
    await supabase.from('units').update({ name, description }).eq('id', id)
  }

  const deleteUnit = async (id: string) => {
    await supabase.from('units').delete().eq('id', id)
  }

  const addBed = async (unitId: string, bedNumber: string) => {
    await supabase.from('beds').insert({ unit_id: unitId, bed_number: bedNumber })
  }

  const updateBed = async (id: string, bedNumber: string) => {
    await supabase.from('beds').update({ bed_number }).eq('id', id)
  }

  const deleteBed = async (id: string) => {
    await supabase.from('beds').delete().eq('id', id)
  }

  const updateEncounter = async (id: string, data: Partial<Encounter>) => {
    await supabase.from('encounters').update(data).eq('id', id)
  }

  const addTask = async (encounterId: string, description: string) => {
    await supabase.from('tasks').insert({ encounter_id: encounterId, description })
  }

  const updateTask = async (id: string, isCompleted: boolean) => {
    await supabase.from('tasks').update({ is_completed: isCompleted }).eq('id', id)
  }

  const deleteTask = async (id: string) => {
    await supabase.from('tasks').delete().eq('id', id)
  }

  return (
    <BoardContext.Provider
      value={{
        data,
        loading,
        addUnit,
        updateUnit,
        deleteUnit,
        addBed,
        updateBed,
        deleteBed,
        updateEncounter,
        addTask,
        updateTask,
        deleteTask,
      }}
    >
      {children}
    </BoardContext.Provider>
  )
}

export const useBoard = () => {
  const context = useContext(BoardContext)
  if (context === undefined) {
    throw new Error('useBoard must be used within a BoardProvider')
  }
  return context
}

  encounter: any
  tasks: any[]
  unit: any

  const [editData, setEditData] = useState<any>({})
  const [newTask, setNewTask] = useState<{ [key: string]: string }>({})

  useEffect(() => {
    const loadNurses = async () => {
      const { data } = await getProfile('') // This is a hack, better to fetch all profiles
      // Actually, let's fetch all profiles from supabase directly in the component or context
      // For simplicity, we'll fetch them here
      const { data: profiles } = await (await import('@/lib/supabase')).supabase
        .from('profiles')
        .select('*')
        .eq('role', 'nurse')
        .eq('is_active', true)
      if (profiles) setNurses(profiles)
    }
    loadNurses()
  }, [])

  if (loading) return <div className="text-center py-10">Loading Board...</div>

  // Build bed data with encounters and tasks
  const bedsWithEncounters: BedWithEncounter[] = data.beds
    .filter((bed) => bed.is_active)
    .map((bed) => {
      const encounter = data.encounters.find((e) => e.bed_id === bed.id)
      const tasks = data.tasks.filter((t) => t.encounter_id === encounter?.id)
      const unit = data.units.find((u) => u.id === bed.unit_id)
      return { bed, encounter, tasks, unit }
    })

  // Group by unit
  const unitsWithBeds = data.units.map((unit) => ({
    unit,
    beds: bedsWithEncounters.filter((b) => b.unit?.id === unit.id),
  }))

    await updateEncounter(id, editData)
    setEditingEncounter(null)
  }

    if (newTask[encounterId]) {
      await addTask(encounterId, newTask[encounterId])
      setNewTask({ ...newTask, [encounterId]: '' })
    }
  }

  const handleToggleTask = async (taskId: string, isCompleted: boolean) => {
    await updateTask(taskId, !isCompleted)
  }

  const handleDeleteTask = async (taskId: string) => {
    await deleteTask(taskId)
  }

  return (
    <div className="space-y-6">
      <h2 className="text-2xl font-bold text-gray-800">Patient Board</h2>

'use client'

import { useState, useEffect, useRef } from 'react'
import { supabase } from '@/lib/supabase'
import { useAuth } from '@/contexts/AuthContext'
import { Send } from 'lucide-react'

export default function ChatRoom() {
  const { user, profile } = useAuth()
  const [messages, setMessages] = useState<any[]>([])
  const [newMessage, setNewMessage] = useState('')
  const messagesEndRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (!user) return

    const loadMessages = async () => {
      const { data } = await supabase
        .from('chat_messages')
        .select('*')
        .order('created_at', { ascending: true })
        .limit(50)
      if (data) setMessages(data)
    }

    loadMessages()

    const channel = supabase
      .channel('chat_channel')
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'chat_messages',
        },
        (payload) => {
          setMessages((prev) => [...prev, payload.new])
        }
      )
      .subscribe()

    return () => {
      channel.unsubscribe()
    }
  }, [user])

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
  }, [messages])

  const handleSendMessage = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!newMessage.trim() || !user) return

    await supabase.from('chat_messages').insert({
      user_id: user.id,
      message: newMessage,
    })

    setNewMessage('')
  }

  return (
    <div className="bg-white rounded-lg shadow-md p-4 h-96 flex flex-col">
      <h3 className="text-lg font-bold text-gray-800 mb-4">Public Forum</h3>
      <div className="flex-1 overflow-y-auto chat-scroll space-y-2 mb-4">
        {messages.map((msg) => {
          const sender = profile?.id === msg.user_id ? profile : null
          // In a real app, you'd fetch the sender's profile for each message
          // For simplicity, we'll just show the user_id if not the current user
          return (
            <div key={msg.id} className="bg-gray-50 p-2 rounded">
              <div className="flex justify-between items-center mb-1">
                <span className="font-semibold text-sm text-primary">
                  {sender?.full_name || 'Unknown User'} ({sender?.role || 'User'})
                </span>
                <span className="text-xs text-gray-500">
                  {new Date(msg.created_at).toLocaleTimeString()}
                </span>
              </div>
              <p className="text-sm text-gray-700">{msg.message}</p>
            </div>
          )
        })}
        <div ref={messagesEndRef} />
      </div>
      <form onSubmit={handleSendMessage} className="flex space-x-2">
        <input
          type="text"
          value={newMessage}
          onChange={(e) => setNewMessage(e.target.value)}
          className="flex-1 p-2 border rounded focus:outline-none focus:ring-2 focus:ring-primary"
          placeholder="Type a message..."
        />
        <button
          type="submit"
          className="bg-primary text-white p-2 rounded hover:bg-blue-700"
        >
          <Send size={18} />
        </button>
      </form>
    </div>
  )
}
