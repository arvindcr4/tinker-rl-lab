Here is the complete implementation for the multiplayer vocabulary-drawing game.

### Technical Architecture
*   **Framework**: Next.js 14+ (App Router).
*   **Language**: TypeScript.
*   **Database/Auth**: Supabase (PostgreSQL, Auth, Realtime).
*   **Real-time**: Supabase Realtime (for game state, chat, and lobby updates).
*   **Drawing**: HTML5 Canvas with `pointer-events` for cross-device compatibility.
*   **Styling**: Tailwind CSS.
*   **Icons**: Lucide React.

### Prerequisites
You must create a Supabase project and run the SQL schema provided in `supabase_schema.sql` in your Supabase SQL Editor. You must also set the environment variables listed in `.env.local`.

-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- 1. Users Profile (extends Supabase Auth)
create table profiles (
  id uuid references auth.users not null primary key,
  username text unique not null,
  score int default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Game Rooms
create table game_rooms (
  id uuid default uuid_generate_v4() primary key,
  name text not null,
  host_id uuid references auth.users(id),
  status text default 'waiting' check (status in ('waiting', 'playing', 'finished')),
  max_players int default 6,
  current_players int default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Room Players
create table room_players (
  id uuid default uuid_generate_v4() primary key,
  room_id uuid references game_rooms(id) on delete cascade,
  user_id uuid references auth.users(id),
  is_drawing boolean default false,
  score int default 0,
  joined_at timestamp with time zone default timezone('utc'::text, now())
);

-- 4. Game Rounds (One per turn)
create table game_rounds (
  id uuid default uuid_generate_v4() primary key,
  room_id uuid references game_rooms(id) on delete cascade,
  word text not null,
  drawer_id uuid references auth.users(id),
  started_at timestamp with time zone default timezone('utc'::text, now()),
  finished_at timestamp with time zone
);

-- 5. Drawing Strokes (Replay Data)
create table drawing_strokes (
  id uuid default uuid_generate_v4() primary key,
  round_id uuid references game_rounds(id) on delete cascade,
  stroke_data jsonb not null, -- { points: [{x, y}], color, size, timestamp }
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 6. Chat / Guesses
create table game_messages (
  id uuid default uuid_generate_v4() primary key,
  room_id uuid references game_rooms(id) on delete cascade,
  round_id uuid references game_rounds(id) on delete cascade,
  user_id uuid references auth.users(id),
  message text not null,
  is_guess boolean default false,
  is_correct boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- Row Level Security (RLS) Policies
alter table profiles enable row level security;
alter table game_rooms enable row level security;
alter table room_players enable row level security;
alter table game_rounds enable row level security;
alter table drawing_strokes enable row level security;
alter table game_messages enable row level security;

-- Profiles: Users can read own profile
create policy "Public profiles are viewable by everyone."
  on profiles for select
  using ( true );

create policy "Users can insert own profile."
  on profiles for insert
  with check ( auth.uid() = id );

create policy "Users can update own profile."
  on profiles for update
  using ( auth.uid() = id );

-- Game Rooms: Anyone can view, only host can update status
create policy "Rooms are viewable by everyone."
  on game_rooms for select
  using ( true );

create policy "Host can update room."
  on game_rooms for update
  using ( auth.uid() = host_id );

-- Room Players: Viewable by room members
create policy "Room players viewable by room members."
  on room_players for select
  using ( 
    exists (
      select 1 from game_rooms 
      where game_rooms.id = room_players.room_id 
      and game_rooms.status != 'finished'
    )
  );

-- Strokes: Viewable by room members
create policy "Strokes viewable by room members."
  on drawing_strokes for select
  using ( 
    exists (
      select 1 from game_rounds gr
      join game_rooms grm on grm.id = gr.room_id
      where grm.id = (select room_id from room_players where user_id = auth.uid())
    )
  );

-- Messages: Viewable by room members
create policy "Messages viewable by room members."
  on game_messages for select
  using ( 
    exists (
      select 1 from game_rooms grm
      where grm.id = game_messages.room_id
      and grm.status != 'finished'
    )
  );

-- Trigger to create profile on signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, username, score)
  values (new.id, split_part(new.email, '@', 1), 0);
  return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

'use client'

import { useEffect, useState } from 'react'
import { createClient } from '@/lib/supabase/client'
import { useRouter } from 'next/navigation'
import { Gamepad2, Plus, LogOut } from 'lucide-react'

interface Room {
  id: string
  name: string
  current_players: number
  max_players: number
  status: string
}

export default function Lobby() {
  const [rooms, setRooms] = useState<Room[]>([])
  const [newRoomName, setNewRoomName] = useState('')
  const [loading, setLoading] = useState(true)
  const supabase = createClient()
  const router = useRouter()

  const fetchRooms = async () => {
    const { data, error } = await supabase
      .from('game_rooms')
      .select('*')
      .order('created_at', { ascending: false })
    
    if (!error && data) {
      setRooms(data)
    }
    setLoading(false)
  }

  useEffect(() => {
    fetchRooms()
    
    const channel = supabase.channel('public:game_rooms')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'game_rooms' }, (payload) => {
        fetchRooms()
      })
      .subscribe()

    return () => {
      supabase.removeChannel(channel)
    }
  }, [supabase])

  const handleCreateRoom = async () => {
    if (!newRoomName.trim()) return
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return

    const { error } = await supabase.from('game_rooms').insert({
      name: newRoomName,
      host_id: user.id,
      max_players: 6
    })

    if (!error) {
      setNewRoomName('')
      fetchRooms()
    }
  }

  const handleJoinRoom = (roomId: string) => {
    router.push(`/game/${roomId}`)
  }

  const handleLogout = async () => {
    await supabase.auth.signOut()
    router.refresh()
  }

  if (loading) return <div className="min-h-screen flex items-center justify-center">Loading...</div>

  return (
    <div className="min-h-screen bg-surface p-4">
      <div className="max-w-4xl mx-auto">
        <div className="flex justify-between items-center mb-8">
          <h1 className="text-3xl font-bold text-gray-800 flex items-center gap-2">
            <Gamepad2 className="text-primary" /> Draw & Guess Lobby
          </h1>
          <button onClick={handleLogout} className="flex items-center gap-2 text-gray-600 hover:text-danger">
            <LogOut size={20} /> Logout
          </button>
        </div>

        <div className="bg-white p-6 rounded-lg shadow-md mb-8">
          <h2 className="text-xl font-semibold mb-4">Create New Room</h2>
          <div className="flex gap-2">
            <input
              type="text"
              value={newRoomName}
              onChange={(e) => setNewRoomName(e.target.value)}
              placeholder="Room Name"
              className="flex-1 p-2 border rounded-md focus:outline-none focus:ring-2 focus:ring-primary"
            />
            <button
              onClick={handleCreateRoom}
              className="bg-primary text-white px-4 py-2 rounded-md hover:bg-indigo-700 flex items-center gap-2"
            >
              <Plus size={20} /> Create
            </button>
          </div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
          {rooms.map((room) => (
            <div key={room.id} className={`bg-white p-4 rounded-lg shadow-md border-l-4 ${room.status === 'playing' ? 'border-yellow-400' : 'border-green-500'}`}>
              <div className="flex justify-between items-start mb-2">
                <h3 className="font-bold text-lg truncate">{room.name}</h3>
                <span className={`text-xs px-2 py-1 rounded-full ${room.status === 'playing' ? 'bg-yellow-100 text-yellow-800' : 'bg-green-100 text-green-800'}`}>
                  {room.status === 'playing' ? 'In Progress' : 'Waiting'}
                </span>
              </div>
              <p className="text-gray-600 text-sm mb-4">
                Players: {room.current_players} / {room.max_players}
              </p>
              <button
                onClick={() => handleJoinRoom(room.id)}
                disabled={room.status === 'playing' || room.current_players >= room.max_players}
                className={`w-full py-2 rounded-md font-medium transition ${
                  room.status === 'playing' || room.current_players >= room.max_players
                    ? 'bg-gray-200 text-gray-500 cursor-not-allowed'
                    : 'bg-primary text-white hover:bg-indigo-700'
                }`}
              >
                {room.status === 'playing' ? 'In Progress' : room.current_players >= room.max_players ? 'Full' : 'Join'}
              </button>
            </div>
          ))}
        </div>
      </div>
    </div>
  )
}

'use client'

import { useEffect, useState, useRef } from 'react'
import { createClient } from '@/lib/supabase/client'
import { useRouter } from 'next/navigation'
import { LogOut, User, Users, MessageSquare, Play, Pause, RotateCcw, Eye } from 'lucide-react'
import DrawingCanvas from './DrawingCanvas'
import Chat from './Chat'
import PlayerList from './PlayerList'
import ReplayViewer from './ReplayViewer'

interface Player {
  id: string
  user_id: string
  username: string
  score: number
  is_drawing: boolean
}

interface Room {
  id: string
  name: string
  status: string
}

interface Round {
  id: string
  word: string
  drawer_id: string
  started_at: string
  finished_at: string | null
}

export default function GameRoom({ roomId }: { roomId: string }) {
  const [room, setRoom] = useState<Room | null>(null)
  const [players, setPlayers] = useState<Player[]>([])
  const [currentRound, setCurrentRound] = useState<Round | null>(null)
  const [myPlayer, setMyPlayer] = useState<Player | null>(null)
  const [viewMode, setViewMode] = useState<'play' | 'replay'>('play')
  const [replayRoundId, setReplayRoundId] = useState<string | null>(null)
  
  const supabase = createClient()
  const router = useRouter()
  const [user, setUser] = useState<any>(null)

  useEffect(() => {
    supabase.auth.getUser().then(({ data }) => {
      if (data.user) setUser(data.user)
      else router.push('/')
    })
  }, [supabase, router])

  // Fetch initial data
  useEffect(() => {
    const fetchData = async () => {
      const { data: roomData } = await supabase.from('game_rooms').select('*').eq('id', roomId).single()
      setRoom(roomData)

      const { data: playersData } = await supabase
        .from('room_players')
        .select(`
          *,
          profiles ( username )
        `)
        .eq('room_id', roomId)
      
      if (playersData) {
        setPlayers(playersData.map((p: any) => ({
          ...p,
          username: p.profiles?.username || 'Unknown'
        })))
        
        const me = playersData.find((p: any) => p.user_id === user?.id)
        setMyPlayer(me)
      }

      // Get active round
      const { data: roundData } = await supabase
        .from('game_rounds')
        .select('*')
        .eq('room_id', roomId)
        .is('finished_at', null)
        .order('started_at', { ascending: false })
        .limit(1)
        .single()
      
      if (roundData) setCurrentRound(roundData)
    }

    fetchData()
  }, [roomId, supabase, user])

  // Realtime Listeners
  useEffect(() => {
    if (!user) return

    // Room Status
    const roomChannel = supabase.channel(`room:${roomId}`)
      .on('postgres_changes', { event: 'UPDATE', schema: 'public', table: 'game_rooms' }, (payload) => {
        setRoom(payload.new as any)
      })
      .subscribe()

    // Players
    const playerChannel = supabase.channel(`players:${roomId}`)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'room_players' }, (payload) => {
        // Re-fetch to ensure consistency
        supabase.from('room_players').select(`
          *,
          profiles ( username )
        `).eq('room_id', roomId).then(({ data }) => {
          if (data) {
            setPlayers(data.map((p: any) => ({
              ...p,
              username: p.profiles?.username || 'Unknown'
            })))
            const me = data.find((p: any) => p.user_id === user.id)
            setMyPlayer(me)
          }
        })
      })
      .subscribe()

    // Rounds
    const roundChannel = supabase.channel(`rounds:${roomId}`)
      .on('postgres_changes', { event: 'UPDATE', schema: 'public', table: 'game_rounds' }, (payload) => {
        setCurrentRound(payload.new as any)
        if (payload.new.finished_at) {
          // Round ended, switch to replay view automatically
          setReplayRoundId(payload.new.id)
          setViewMode('replay')
        }
      })
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'game_rounds' }, (payload) => {
        setCurrentRound(payload.new as any)
        setViewMode('play')
      })
      .subscribe()

    return () => {
      supabase.removeChannel(roomChannel)
      supabase.removeChannel(playerChannel)
      supabase.removeChannel(roundChannel)
    }
  }, [roomId, supabase, user])

  const handleLogout = async () => {
    await supabase.auth.signOut()
    router.push('/')
  }

  if (!room || !myPlayer) return <div className="min-h-screen flex items-center justify-center">Loading Game...</div>

  const isHost = room.host_id === user?.id
  const isDrawing = myPlayer.is_drawing
  const isGuesser = !isDrawing && room.status === 'playing'

  return (
    <div className="min-h-screen bg-surface flex flex-col">
      {/* Header */}
      <header className="bg-white shadow-sm p-4 flex justify-between items-center z-10">
        <div>
          <h1 className="text-xl font-bold text-gray-800">{room.name}</h1>
          <span className={`text-xs px-2 py-1 rounded-full ${room.status === 'playing' ? 'bg-green-100 text-green-800' : 'bg-yellow-100 text-yellow-800'}`}>
            {room.status === 'playing' ? 'Playing' : 'Waiting'}
          </span>
        </div>
        <div className="flex items-center gap-4">
          {room.status === 'playing' && (

'use client'

import { useEffect, useRef, useState, useCallback } from 'react'
import { createClient } from '@/lib/supabase/client'
import { COLORS } from '@/lib/constants'
import { Undo2, Trash2, Check } from 'lucide-react'

interface DrawingCanvasProps {
  roomId: string
  roundId: string
  isDrawing: boolean
  word: string
}

export default function DrawingCanvas({ roomId, roundId, isDrawing, word }: DrawingCanvasProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const [isDrawingState, setIsDrawingState] = useState(false)
  const [currentStroke, setCurrentStroke] = useState<any>(null)
  const [color, setColor] = useState(COLORS[0])
  const [size, setSize] = useState(5)
  const supabase = createClient()
  const strokesRef = useRef<any[]>([])

  // Initialize Canvas
  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return

    const ctx = canvas.getContext('2d')
    if (!ctx) return

    // Set canvas size to container
    const resize = () => {
      const parent = canvas.parentElement
      if (parent) {
        canvas.width = parent.clientWidth
        canvas.height = parent.clientHeight
        redraw()
      }
    }
    
    window.addEventListener('resize', resize)
    resize()

    return () => window.removeEventListener('resize', resize)
  }, [])

  // Redraw all strokes
  const redraw = useCallback(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    if (!ctx) return

    ctx.clearRect(0, 0, canvas.width, canvas.height)
    ctx.fillStyle = '#FFFFFF'
    ctx.fillRect(0, 0, canvas.width, canvas.height)

    strokesRef.current.forEach((stroke: any) => {
      drawStroke(ctx, stroke)
    })
  }, [])

  const drawStroke = (ctx: CanvasRenderingContext2D, stroke: any) => {
    if (stroke.points.length < 2) return
    ctx.beginPath()
    ctx.strokeStyle = stroke.color
    ctx.lineWidth = stroke.size
    ctx.lineCap = 'round'
    ctx.lineJoin = 'round'
    ctx.moveTo(stroke.points[0].x, stroke.points[0].y)
    for (let i = 1; i < stroke.points.length; i++) {
      ctx.lineTo(stroke.points[i].x, stroke.points[i].y)
    }
    ctx.stroke()
  }

  // Load existing strokes for replay or ongoing game
  useEffect(() => {
    if (!roundId) return

    const loadStrokes = async () => {
      const { data } = await supabase
        .from('drawing_strokes')
        .select('*')
        .eq('round_id', roundId)
        .order('created_at', { ascending: true })
      
      if (data) {
        strokesRef.current = data
        redraw()
      }
    }

    loadStrokes()

    // Listen for new strokes
    const channel = supabase.channel(`strokes:${roundId}`)
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'drawing_strokes' }, (payload) => {
        strokesRef.current.push(payload.new)
        redraw()
      })
      .subscribe()

    return () => {
      supabase.removeChannel(channel)
    }
  }, [roundId, supabase, redraw])

  const startDrawing = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (!isDrawing) return
    setIsDrawingState(true)
    const canvas = canvasRef.current
    if (!canvas) return
    const rect = canvas.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top

    setCurrentStroke({
      points: [{ x, y }],
      color,
      size,
      timestamp: Date.now()
    })
  }

  const draw = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (!isDrawingState || !isDrawing || !currentStroke) return
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    if (!ctx) return

    const rect = canvas.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top

    currentStroke.points.push({ x, y })
    drawStroke(ctx, currentStroke)
  }

  const stopDrawing = async () => {
    if (!isDrawingState || !currentStroke || !roundId) return
    setIsDrawingState(false)
    
    // Save to DB
    await supabase.from('drawing_strokes').insert({
      round_id: roundId,
      stroke_data: currentStroke
    })

    setCurrentStroke(null)
  }

  const handleUndo = async () => {
    if (strokesRef.current.length === 0) return
    const lastStroke = strokesRef.current.pop()
    if (lastStroke) {
      await supabase.from('drawing_strokes').delete().eq('id', lastStroke.id)
      redraw()
    }
  }

  const handleClear = async () => {
    if (!roundId) return
    await supabase.from('drawing_strokes').delete().eq('round_id', roundId)
    strokesRef.current = []
    redraw()
  }

  const handleSubmitWord = async () => {
    // Logic to end round could go here, but usually handled by timer or host
    alert("Round will end when time is up or host ends it.")
  }

  return (
    <div className="relative w-full h-full bg-white">
      {/* Word Display (Only for Drawer) */}
      {isDrawing && word && (
        <div className="absolute top-4 left-1/2 transform -translate-x-1/2 bg-white shadow-lg px-6 py-2 rounded-full z-10 border-2 border-primary">
          <span className="text-xl font-bold text-gray-800">{word}</span>
        </div>
      )}

      {/* Toolbar */}
      {isDrawing && (
        <div className="absolute bottom-4 left-1/2 transform -translate-x-1/2 bg-white shadow-lg p-2 rounded-full flex items-center gap-2 z-10">
          {COLORS.map(c => (
            <button

              className={`w-6 h-6 rounded-full border-2 ${color === c ? 'border-gray-800' : 'border-transparent'}`}
              style={{ backgroundColor: c }}
            />
          ))}
          <div className="w-px h-6 bg-gray-300 mx-1" />
          <button onClick={handleUndo} className="p-2 hover:bg-gray-100 rounded-full" title="Undo">
            <Undo2 size={20} />

          <button onClick={handleClear} className="p-2 hover:bg-gray-100 rounded-full text-danger" title="Clear">

  )

'use client'

import { useEffect, useRef, useState } from 'react'
import { createClient } from '@/lib/supabase/client'
import { Play, Pause, RotateCcw } from 'lucide-react'

export default function ReplayViewer({ roundId }: { roundId: string }) {
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const [strokes, setStrokes] = useState<any[]>([])
  const [isPlaying, setIsPlaying] = useState(false)
  const [currentStrokeIndex, setCurrentStrokeIndex] = useState(0)
  const [isPaused, setIsPaused] = useState(false)
  const supabase = createClient()
  const animationRef = useRef<number>()

  useEffect(() => {
    const loadStrokes = async () => {
      const { data } = await supabase
        .from('drawing_strokes')
        .select('*')
        .eq('round_id', roundId)
        .order('created_at', { ascending: true })
      
      if (data) {
        setStrokes(data)
        setCurrentStrokeIndex(0)
        setIsPlaying(true)
      }
    }

    loadStrokes()
  }, [roundId, supabase])

  useEffect(() => {
    if (!isPlaying || isPaused || strokes.length === 0) return

    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    if (!ctx) return

    // Set canvas size
    const parent = canvas.parentElement
    if (parent) {
      canvas.width = parent.clientWidth
      canvas.height = parent.clientHeight
    }

    let index = 0
    let currentPoints = 0
    
    const animate = () => {
      if (index >= strokes.length) {
        setIsPlaying(false)
        return
      }

      const stroke = strokes[index]
      
      // Clear and redraw previous strokes
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      ctx.fillStyle = '#FFFFFF'
      ctx.fillRect(0, 0, canvas.width, canvas.height)

      for (let i = 0; i < index; i++) {
        drawFullStroke(ctx, strokes[i])
      }

      // Draw current stroke progressively
      if (currentPoints < stroke.stroke_data.points.length) {
        currentPoints++
        drawPartialStroke(ctx, stroke, currentPoints)
        animationRef.current = requestAnimationFrame(animate)
      } else {
        index++
        currentPoints = 0
        animationRef.current = requestAnimationFrame(animate)
      }
    }

    animate()

    return () => {
      if (animationRef.current) cancelAnimationFrame(animationRef.current)
    }
  }, [isPlaying, isPaused, strokes])

  const drawFullStroke = (ctx: CanvasRenderingContext2D, stroke: any) => {
    const data = stroke.stroke_data
    if (data.points.length < 2) return
    ctx.beginPath()
    ctx.strokeStyle = data.color
    ctx.lineWidth = data.size
    ctx.lineCap = 'round'
    ctx.lineJoin = 'round'
    ctx.moveTo(data.points[0].x, data.points[0].y)
    for (let i = 1; i < data.points.length; i++) {
      ctx.lineTo(data.points[i].x, data.points[i].y)
    }
    ctx.stroke()
  }

  const drawPartialStroke = (ctx: CanvasRenderingContext2D, stroke: any, count: number) => {
    const data = stroke.stroke_data
    if (data.points.length < 2) return
    ctx.beginPath()
    ctx.strokeStyle = data.color
    ctx.lineWidth = data.size
    ctx.lineCap = 'round'
    ctx.lineJoin = 'round'
    ctx.moveTo(data.points[0].x, data.points[0].y)
    for (let i = 1; i < count; i++) {
      ctx.lineTo(data.points[i].x, data.points[i].y)
    }
    ctx.stroke()
  }

  const handlePlayPause = () => {
    setIsPlaying(!isPlaying)
    if (!isPlaying) setIsPaused(false)
    else setIsPaused(true)
  }

  const handleRestart = () => {
    setIsPlaying(true)
    setIsPaused(false)
    setCurrentStrokeIndex(0)
  }

  return (
    <div className="relative w-full h-full bg-white flex flex-col items-center justify-center">
      <div className="absolute top-4 left-4 z-10 bg-white shadow-lg p-2 rounded-full flex gap-2">
        <button onClick={handlePlayPause} className="p-2 hover:bg-gray-100 rounded-full">
          {isPlaying ? <Pause size={24} /> : <Play size={24} />}
        </button>
        <button onClick={handleRestart} className="p-2 hover:bg-gray-100 rounded-full">
          <RotateCcw size={24} />
        </button>
      </div>
      <canvas ref={canvasRef} className="w-full h-full" />
    </div>
  )
}
