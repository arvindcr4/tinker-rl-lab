Here is the complete implementation for the Hospital Patient Tracking Board.

### Technical Architecture
*   **Framework**: Next.js 14+ (App Router).
*   **Styling**: Tailwind CSS for responsive, hospital-grade UI.
*   **Database & Auth**: Supabase (PostgreSQL, Auth, Realtime).
*   **State Management**: React Context + Supabase JS Client subscriptions for real-time sync.
*   **Icons**: Lucide React.

### Prerequisites
You must have a Supabase project.
1.  Run the SQL provided in `supabase-schema.sql` in your Supabase SQL Editor.
2.  Set the environment variables in your `.env.local` file.

create table profiles (
  id uuid references auth.users on delete cascade primary key,
  full_name text not null,
  role text check (role in ('admin', 'nurse', 'physician')) not null,
  avatar_url text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Units Table
create table units (
  id uuid default uuid_generate_v4() primary key,
  name text not null,
  description text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Beds Table
create table beds (
  id uuid default uuid_generate_v4() primary key,
  unit_id uuid references units(id) on delete cascade not null,
  bed_number text not null,
  is_active boolean default true,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 4. Patients Table
create table patients (
  id uuid default uuid_generate_v4() primary key,
  mrn text not null, -- Medical Record Number
  first_name text not null,
  last_name text not null,
  dob date not null,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 5. Encounters Table (Links Patient to Bed)
create table encounters (
  id uuid default uuid_generate_v4() primary key,
  patient_id uuid references patients(id) on delete cascade not null,
  bed_id uuid references beds(id) on delete cascade not null,
  assigned_nurse_id uuid references profiles(id) on delete set null,
  status text check (status in ('admitted', 'procedure', 'recovery', 'discharge-ready', 'discharged')) default 'admitted',
  notes text default '',
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 6. Tasks Table
create table tasks (
  id uuid default uuid_generate_v4() primary key,
  encounter_id uuid references encounters(id) on delete cascade not null,
  description text not null,
  is_completed boolean default false,
  created_by uuid references profiles(id) on delete set null,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 7. Chat Messages
create table chat_messages (
  id uuid default uuid_generate_v4() primary key,
  user_id uuid references profiles(id) on delete cascade not null,
  message text not null,

);

-- 8. Emergency Alerts
create table emergency_alerts (
  id uuid default uuid_generate_v4() primary key,
  title text not null,
  message text not null,

  created_by uuid references profiles(id) on delete set null,

);

create table user_alert_dismissals (
  user_id uuid references profiles(id) on delete cascade,
  alert_id uuid references emergency_alerts(id) on delete cascade,
  created_at timestamp with time zone default timezone('utc'::text, now()),
  primary key (user_id, alert_id)
);

    process.env.NEXT_PUBLIC_SUPABASE_URL!,

            // The `setAll` method was called from a Server Component.

            // user sessions.
          }

      },
    }
  )

import { type ClassValue, clsx } from 'clsx';
import { twMerge } from 'tailwind-merge';

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

export function calculateAge(dob: string): number {
  const today = new Date();
  const birthDate = new Date(dob);
  let age = today.getFullYear() - birthDate.getFullYear();
  const m = today.getMonth() - birthDate.getMonth();
  if (m < 0 || (m === 0 && today.getDate() < birthDate.getDate())) {
    age--;
  }
  return age;
}

export function getSeverityColor(severity: string): string {
  switch (severity) {
    case 'critical': return 'bg-alert-critical text-white';
    case 'high': return 'bg-alert-high text-white';
    case 'medium': return 'bg-alert-medium text-black';
    case 'low': return 'bg-alert-low text-white';
    default: return 'bg-gray-500 text-white';
  }
}

'use client';

import { createClient } from '@/lib/supabase/client';
import { useEffect, useState, createContext, useContext } from 'react';
import { Session, User } from '@supabase/supabase-js';

const AuthContext = createContext<{
  session: Session | null;
  user: User | null;
  loading: boolean;
}>({
  session: null,
  user: null,
  loading: true,
});

export const useAuth = () => useContext(AuthContext);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);
  const supabase = createClient();

  useEffect(() => {
    const getSession = async () => {
      const { data: { session } } = await supabase.auth.getSession();
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    };

    getSession();

    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    return () => subscription.unsubscribe();
  }, [supabase]);

  return (
    <AuthContext.Provider value={{ session, user, loading }}>
      {children}
    </AuthContext.Provider>
  );
}

'use client';

import { createClient } from '@/lib/supabase/client';
import { useEffect, useState, createContext, useContext } from 'react';
import { Profile } from '@/lib/types';

const PresenceContext = createContext<{
  onlineUsers: Profile[];
}>({
  onlineUsers: [],
});

export const usePresence = () => useContext(PresenceContext);

export function PresenceProvider({ children }: { children: React.ReactNode }) {
  const [onlineUsers, setOnlineUsers] = useState<Profile[]>([]);
  const supabase = createClient();

  useEffect(() => {
    // Listen for profile changes (which includes online status if we added a column, 
    // but here we rely on the fact that profiles are updated on login/logout or we can use a heartbeat).
    // For this implementation, we will assume "Online" is determined by active sessions 
    // or we can add a simple 'last_seen' column. 
    // To keep it simple and robust without complex presence tables:
    // We will just listen to profiles updates. 
    // A more robust way is a 'users_online' table updated by heartbeat.
    
    // Let's implement a simple heartbeat mechanism in the Dashboard component 
    // that updates a 'last_active' timestamp in profiles.
    
    const channel = supabase
      .channel('presence_changes')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'profiles' }, (payload) => {
        // Refetch all profiles to update list
        fetchProfiles();
      })
      .subscribe();

    const fetchProfiles = async () => {
      const { data } = await supabase
        .from('profiles')
        .select('*')
        .order('full_name');
      if (data) setOnlineUsers(data);
    };

    fetchProfiles();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [supabase]);

  return (
    <PresenceContext.Provider value={{ onlineUsers }}>
      {children}
    </PresenceContext.Provider>
  );
}

'use client';

import { createClient } from '@/lib/supabase/client';
import { useEffect, useState, createContext, useContext } from 'react';
import { Alert, Dismissal } from '@/lib/types';
import { useAuth } from './AuthProvider';

const AlertContext = createContext<{
  alerts: Alert[];
  dismissedAlertIds: string[];
  createAlert: (title: string, message: string, severity: Alert['severity']) => Promise<void>;
  dismissAlert: (alertId: string) => Promise<void>;
}>({
  alerts: [],
  dismissedAlertIds: [],
  createAlert: async () => {},
  dismissAlert: async () => {},
});

export const useAlerts = () => useContext(AlertContext);

export function AlertProvider({ children }: { children: React.ReactNode }) {
  const [alerts, setAlerts] = useState<Alert[]>([]);
  const [dismissedAlertIds, setDismissedAlertIds] = useState<string[]>([]);
  const { user } = useAuth();
  const supabase = createClient();

  const fetchAlerts = async () => {
    const { data } = await supabase
      .from('emergency_alerts')
      .select('*')
      .eq('is_active', true)
      .order('created_at', { ascending: false });
    if (data) setAlerts(data);
  };

  const fetchDismissals = async () => {
    if (!user) {
      setDismissedAlertIds([]);
      return;
    }
    const { data } = await supabase
      .from('user_alert_dismissals')
      .select('alert_id')
      .eq('user_id', user.id);
    if (data) setDismissedAlertIds(data.map(d => d.alert_id));
  };

  useEffect(() => {
    fetchAlerts();
    fetchDismissals();

    const channel = supabase
      .channel('alerts_changes')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'emergency_alerts' }, () => {
        fetchAlerts();
        fetchDismissals();
      })
      .on('postgres_changes', { event: '*', schema: 'public', table: 'user_alert_dismissals' }, () => {
        fetchDismissals();
      })
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [supabase, user]);

  const createAlert = async (title: string, message: string, severity: Alert['severity']) => {
    if (!user) return;
    await supabase.from('emergency_alerts').insert({
      title,
      message,
      severity,
      created_by: user.id,
      is_active: true
    });
  };

  const dismissAlert = async (alertId: string) => {
    if (!user) return;
    await supabase.from('user_alert_dismissals').upsert({
      user_id: user.id,
      alert_id: alertId
    });
  };

  return (
    <AlertContext.Provider value={{ alerts, dismissedAlertIds, createAlert, dismissAlert }}>
      {children}
    </AlertContext.Provider>
  );
}

                  <option value="">Select Patient</option>
                  {patients.map(p => (
                    <option key={p.id} value={p.id}>{p.last_name}, {p.first_name} ({p.mrn})</option>
                  ))}
                </select>
                <select
                  value={newEncounterData.bedId}
                  onChange={(e) => setNewEncounterData({...newEncounterData, bedId: e.target.value})}
                  className="border border-gray-300 rounded-md px-3 py-2"
                >
                  <option value="">Select Bed</option>
                  {beds.map(b => {
                    const unit = units.find(u => u.id === b.unit_id);
                    return (
                      <option key={b.id} value={b.id}>{unit?.name} - Bed {b.bed_number}</option>
                    );
                  })}
                </select>
                <select
                  value={newEncounterData.nurseId}
                  onChange={(e) => setNewEncounterData({...newEncounterData, nurseId: e.target.value})}
                  className="border border-gray-300 rounded-md px-3 py-2"
                >
                  <option value="">Select Nurse</option>
                  {onlineUsers.filter(u => u.role === 'nurse').map(n => (
                    <option key={n.id} value={n.id}>{n.full_name}</option>
                  ))}
                </select>
                <select
                  value={newEncounterData.status}
                  onChange={(e) => setNewEncounterData({...newEncounterData, status: e.target.value})}
                  className="border border-gray-300 rounded-md px-3 py-2"
                >
                  <option value="admitted">Admitted</option>
                  <option value="procedure">Procedure</option>
                  <option value="recovery">Recovery</option>
                  <option value="discharge-ready">Discharge Ready</option>
                </select>
                <button type="submit" className="bg-hospital-600 text-white px-4 py-2 rounded-md hover:bg-hospital-700 md:col-span-2">
                  Create Encounter
                </button>
              </form>
            </div>

            {/* Create Alert */}
            <div className="bg-white p-6 rounded-lg shadow">
              <h3 className="text-lg font-medium text-gray-900 mb-4">Create Emergency Alert</h3>
              <form onSubmit={handleCreateAlert} className="space-y-4">
                <div>
                  <label className="block text-sm font-medium text-gray-700">Title</label>
                  <input
                    type="text"
                    value={newAlertData.title}
                    onChange={(e) => setNewAlertData({...newAlertData, title: e.target.value})}
                    className="mt-1 block w-full border border-gray-300 rounded-md px-3 py-2"
                  />
                </div>
                <div>
                  <label className="block text-sm font-medium text-gray-700">Message</label>
                  <textarea
                    value={newAlertData.message}
                    onChange={(e) => setNewAlertData({...newAlertData, message: e.target.value})}
                    className="mt-1 block w-full border border-gray-300 rounded-md px-3 py-2"
                    rows={3}
                  />
                </div>
                <div>
                  <label className="block text-sm font-medium text-gray-700">Severity</label>
                  <select
                    value={newAlertData.severity}
                    onChange={(e) => setNewAlertData({...newAlertData, severity: e.target.value as any})}
                    className="mt-1 block w-full border border-gray-300 rounded-md px-3 py-2"
                  >
                    <option value="low">Low</option>
                    <option value="medium">Medium</option>
                    <option value="high">High</option>
                    <option value="critical">Critical</option>
                  </select>
                </div>
                <button type="submit" className="bg-red-600 text-white px-4 py-2 rounded-md hover:bg-red-700">
                  Broadcast Alert
                </button>
              </form>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
