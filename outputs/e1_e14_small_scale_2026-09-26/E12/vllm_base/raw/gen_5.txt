Here is the complete implementation for the multiplayer vocabulary-drawing game.

### Technical Architecture
*   **Framework**: Next.js 14+ (App Router).
*   **Styling**: Tailwind CSS.
*   **State Management**: Zustand (Client state), React Query (Server state).
*   **Real-time**: Supabase Realtime (WebSockets) for game events, lobby updates, and chat.
*   **Database**: Supabase PostgreSQL.
*   **Drawing**: HTML5 Canvas with `pointer-events` for cross-device compatibility.
*   **Replay**: Custom hook that replays stroke data with timing logic.

### Prerequisites
You must create a Supabase project and run the SQL provided in `supabase_schema.sql` in your SQL editor. You must then set the environment variables in your `.env.local` file.

```sql
-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- 1. Users Table (Extends Supabase Auth)
create table public.profiles (
  id uuid references auth.users not null primary key,
  username text unique not null,
  score int default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Games Table
create table public.games (
  id uuid default uuid_generate_v4() primary key,
  name text not null,
  host_id uuid references public.profiles(id),
  status text default 'waiting' check (status in ('waiting', 'playing', 'finished')),
  max_players int default 6,
  current_word text,
  drawing_player_id uuid references public.profiles(id),
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Game Players Table
create table public.game_players (
  id uuid default uuid_generate_v4() primary key,
  game_id uuid references public.games(id) on delete cascade,
  user_id uuid references public.profiles(id),
  score int default 0,
  is_drawing boolean default false,
  joined_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 4. Game Rounds (Replay Data)
create table public.game_rounds (
  id uuid default uuid_generate_v4() primary key,
  game_id uuid references public.games(id) on delete cascade,
  word text not null,
  drawing_player_id uuid references public.profiles(id),
  strokes jsonb not null, -- Array of {x, y, color, size, timestamp}
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 5. Real-time Chat/Guesses
create table public.game_messages (
  id uuid default uuid_generate_v4() primary key,
  game_id uuid references public.games(id) on delete cascade,
  user_id uuid references public.profiles(id),
  message text not null,
  is_guess boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Row Level Security (RLS) Policies
alter table public.profiles enable row level security;
create policy "Public profiles are viewable by everyone." on public.profiles for select using (true);
create policy "Users can update own profile." on public.profiles for update using (auth.uid() = id);

alter table public.games enable row level security;
create policy "Games are viewable by everyone." on public.games for select using (true);
create policy "Users can create games." on public.games for insert with check (auth.uid() = host_id);

alter table public.game_players enable row level security;
create policy "Game players viewable by game members." on public.game_players for select using (
  exists (select 1 from public.games where games.id = game_players.game_id)
);
create policy "Users can join own games." on public.game_players for insert with check (auth.uid() = user_id);

alter table public.game_rounds enable row level security;
create policy "Rounds viewable by game members." on public.game_rounds for select using (
  exists (select 1 from public.games where games.id = game_rounds.game_id)
);

alter table public.game_messages enable row level security;
create policy "Messages viewable by game members." on public.game_messages for select using (
  exists (select 1 from public.games where games.id = game_messages.game_id)
);
create policy "Users can post messages." on public.game_messages for insert with check (auth.uid() = user_id);

-- Triggers for updated_at
create or replace function public.handle_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

create trigger handle_games_updated_at
  before update on public.games
  for each row
  execute procedure public.handle_updated_at();

-- Initial Seed Data for Vocabulary (Elementary Level)
insert into public.games (name, host_id, status) 
select 'Open Lobby', auth.uid(), 'waiting' 
from auth.users 
limit 1;
```

```typescript
export function getRandomWord(exclude: string[] = []): string {
  const available = VOCABULARY.filter(w => !exclude.includes(w));
  if (available.length === 0) return VOCABULARY[Math.floor(Math.random() * VOCABULARY.length)];
```

```typescript
import { useEffect, useRef, useState, useCallback } from 'react';
import { supabase } from './supabase';
import { Database } from './supabase';

type Game = Database['public']['Tables']['games']['Row'];
type GamePlayer = Database['public']['Tables']['game_players']['Row'];
type Message = Database['public']['Tables']['game_messages']['Row'];
type Round = Database['public']['Tables']['game_rounds']['Row'];

export function useAuth() {
  const [user, setUser] = useState<any>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setUser(session?.user ?? null);
      setLoading(false);
    });

    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setUser(session?.user ?? null);
    });

    return () => subscription.unsubscribe();
  }, []);

  const login = async (email: string, password: string) => {
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    if (error) throw error;
  };

  const register = async (email: string, password: string, username: string) => {
    const { data, error } = await supabase.auth.signUp({ email, password });
    if (error) throw error;
    
    if (data.user) {
      await supabase.from('profiles').insert({ id: data.user.id, username });
    }
  };

  const logout = async () => {
    await supabase.auth.signOut();
  };

  return { user, loading, login, register, logout };
}

export function useGame(gameId: string | null) {
  const [game, setGame] = useState<Game | null>(null);
  const [players, setPlayers] = useState<GamePlayer[]>([]);
  const [messages, setMessages] = useState<Message[]>([]);
  const [rounds, setRounds] = useState<Round[]>([]);
  const [currentStrokes, setCurrentStrokes] = useState<any[]>([]);

  useEffect(() => {
    if (!gameId) return;

    const fetchInitialData = async () => {
      const { data: g } = await supabase.from('games').select('*').eq('id', gameId).single();
      setGame(g);

      const { data: p } = await supabase.from('game_players').select('*').eq('game_id', gameId);
      setPlayers(p || []);

      const { data: m } = await supabase.from('game_messages').select('*').eq('game_id', gameId).order('created_at', { ascending: true });
      setMessages(m || []);

      const { data: r } = await supabase.from('game_rounds').select('*').eq('game_id', gameId).order('created_at', { ascending: false });
      setRounds(r || []);
    };

    fetchInitialData();

    const gameChannel = supabase.channel(`game:${gameId}`)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'games', filter: `id=eq.${gameId}` }, (payload) => {
        setGame(payload.new as Game);
      })
      .on('postgres_changes', { event: '*', schema: 'public', table: 'game_players', filter: `game_id=eq.${gameId}` }, (payload) => {
        if (payload.eventType === 'INSERT') {
          setPlayers(prev => [...prev, payload.new as GamePlayer]);
        } else if (payload.eventType === 'DELETE') {
          setPlayers(prev => prev.filter(p => p.id !== payload.old.id));
        } else if (payload.eventType === 'UPDATE') {
          setPlayers(prev => prev.map(p => p.id === payload.new.id ? payload.new as GamePlayer : p));
        }
      })
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'game_messages', filter: `game_id=eq.${gameId}` }, (payload) => {
        setMessages(prev => [...prev, payload.new as Message]);
      })
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'game_rounds', filter: `game_id=eq.${gameId}` }, (payload) => {
        setRounds(prev => [payload.new as Round, ...prev]);
      })
      .subscribe();

    return () => {
      supabase.removeChannel(gameChannel);
    };
  }, [gameId]);

  const sendGuess = async (message: string) => {
    if (!gameId) return;
    await supabase.from('game_messages').insert({ game_id: gameId, message, is_guess: true });
  };

  const saveRound = async (strokes: any[], word: string, drawingPlayerId: string) => {
    if (!gameId) return;
    await supabase.from('game_rounds').insert({
      game_id: gameId,
      word,
      drawing_player_id: drawingPlayerId,
      strokes
    });
  };

  return { game, players, messages, rounds, currentStrokes, setCurrentStrokes, sendGuess, saveRound };
}

export function useReplay(round: Round | null) {
  const [isPlaying, setIsPlaying] = useState(false);
  const [progress, setProgress] = useState(0);
  const [currentStrokeIndex, setCurrentStrokeIndex] = useState(0);
  const animationRef = useRef<number>();
  const startTimeRef = useRef<number>();

  const play = useCallback(() => {
    if (!round || !round.strokes.length) return;
    setIsPlaying(true);
    startTimeRef.current = performance.now();
    
    const animate = (time: number) => {
      if (!startTimeRef.current) return;
      const elapsed = time - startTimeRef.current;
      
      // Calculate total duration based on original timestamps
      const totalDuration = round.strokes[round.strokes.length - 1].timestamp - round.strokes[0].timestamp;
      const currentProgress = Math.min(elapsed, totalDuration);
      
      setProgress(currentProgress);
      
      // Find which stroke should be visible
      const visibleStrokeIndex = round.strokes.findIndex(s => s.timestamp > currentProgress);
      const index = visibleStrokeIndex === -1 ? round.strokes.length - 1 : visibleStrokeIndex - 1;
      
      setCurrentStrokeIndex(index);
      
      if (currentProgress < totalDuration) {
        animationRef.current = requestAnimationFrame(animate);
      } else {
        setIsPlaying(false);
      }
    };
    
    animationRef.current = requestAnimationFrame(animate);
  }, [round]);

  const pause = useCallback(() => {
    setIsPlaying(false);
    if (animationRef.current) cancelAnimationFrame(animationRef.current);
  }, []);

  const restart = useCallback(() => {
    pause();
    setProgress(0);
    setCurrentStrokeIndex(0);
  }, [pause]);

  useEffect(() => {
    return () => {
      if (animationRef.current) cancelAnimationFrame(animationRef.current);
    };
  }, []);

  return { isPlaying, progress, currentStrokeIndex, play, pause, restart };
}
```

```tsx
function AuthForm() {
  const { login, register } = useAuth();
  const [isLogin, setIsLogin] = useState(true);
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');

      setError(err.message);
```

```tsx
  const [showReplay, setShowReplay] = useState(false);

  const canvasRef = useRef<HTMLCanvasElement>(null);
  const isDrawing = useRef(false);
  const lastPos = useRef<{x: number, y: number} | null>(null);

  useEffect(() => {
    if (!user) router.push('/');
    if (!game) return;

    // Check if user is in this game
    const isPlayer = players.some(p => p.user_id === user.id);
    if (!isPlayer && game.status === 'waiting') {
      // Auto-join if waiting
      supabase.from('game_players').insert({
        game_id: params.id,
        user_id: user.id
      });
    } else if (!isPlayer) {
      router.push('/lobby');
    }
  }, [user, game, players, params.id, router]);

  // Canvas Drawing Logic
  const startDrawing = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (!game?.drawing_player_id || game.drawing_player_id !== user?.id) return;
    isDrawing.current = true;
    const { x, y } = getCanvasCoords(e);
    lastPos.current = { x, y };
    
    // Send first point
    const stroke = { x, y, color: '#000000', size: 5, timestamp: Date.now() };
    setCurrentStrokes(prev => [...prev, stroke]);
  };

  const draw = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (!isDrawing.current || !lastPos.current) return;
    const { x, y } = getCanvasCoords(e);
    
    const ctx = canvasRef.current?.getContext('2d');
    if (ctx) {
      ctx.beginPath();
      ctx.moveTo(lastPos.current.x, lastPos.current.y);
      ctx.lineTo(x, y);
      ctx.strokeStyle = '#000000';
      ctx.lineWidth = 5;
      ctx.lineCap = 'round';
      ctx.stroke();
    }

    const stroke = { x, y, color: '#000000', size: 5, timestamp: Date.now() };
    setCurrentStrokes(prev => [...prev, stroke]);
    lastPos.current = { x, y };
  };

  const stopDrawing = () => {
    isDrawing.current = false;
    lastPos.current = null;
  };

  const getCanvasCoords = (e: React.PointerEvent<HTMLCanvasElement>) => {
    const rect = canvasRef.current?.getBoundingClientRect();
    if (!rect) return { x: 0, y: 0 };
    return {
      x: e.clientX - rect.left,
      y: e.clientY - rect.top
    };
  };

  // Handle Guess
  const handleGuess = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!guess.trim() || !game?.current_word) return;
    
    await sendGuess(guess);
    
    if (guess.toLowerCase().trim() === game.current_word.toLowerCase()) {
      // Correct guess logic would update score in DB via trigger or RPC
      // For now, we just clear input
      setGuess('');
    } else {
      setGuess('');
    }
  };

  // End Round Simulation (For demo purposes, we'll end after 60s or manually)
  const endRound = async () => {
    if (!game || !user) return;
    
    // Save round data
    await saveRound(currentStrokes, game.current_word || '', game.drawing_player_id || '');

    setShowReplay(true);

          <span className="font-medium">Word: {isDrawingPlayer ? game.current_word : '???'}</span>
          <button onClick={logout} className="text-red-500 hover:underline">Logout</button>
        </div>
      </header>

      <div className="flex flex-1 overflow-hidden">
        {/* Sidebar: Players & Chat */}
        <aside className="w-80 bg-white border-r flex flex-col">
          <div className="p-4 border-b">
            <h2 className="font-bold mb-2">Players ({players.length})</h2>
            <ul className="space-y-2">
              {players.map(p => (
                <li key={p.id} className="flex justify-between">
                  <span>{p.user_id === user.id ? 'You' : 'Player'}</span>
                  <span className="font-bold">{p.score}</span>
                </li>
              ))}
            </ul>
          </div>
          
          <div className="flex-1 overflow-y-auto p-4">
            <h2 className="font-bold mb-2">Chat</h2>

            <input

              value={guess}
              onChange={(e) => setGuess(e.target.value)}

            <button type="submit" className="bg-primary text-white px-4 py-2 rounded">Send</button>

          <canvas
            ref={canvasRef}
            width={800}
            height={600}
            className="bg-white shadow rounded cursor-crosshair"
            onPointerDown={startDrawing}
            onPointerMove={draw}
            onPointerUp={stopDrawing}
            onPointerLeave={stopDrawing}

function ReplayModal({ round, onClose }: { round: any, onClose: () => void }) {
  const { isPlaying, progress, currentStrokeIndex, play, pause, restart } = useReplay(round);
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const ctx = canvasRef.current?.getContext('2d');
    if (!ctx || !round.strokes.length) return;

    // Clear canvas
    ctx.clearRect(0, 0, 800, 600);

    // Draw all strokes up to current index
    for (let i = 0; i <= currentStrokeIndex; i++) {
      const stroke = round.strokes[i];
      if (i === 0) {
        ctx.beginPath();
        ctx.moveTo(stroke.x, stroke.y);
      } else {
        const prev = round.strokes[i-1];
        ctx.beginPath();
        ctx.moveTo(prev.x, prev.y);
        ctx.lineTo(stroke.x, stroke.y);
      }
      ctx.strokeStyle = stroke.color;
      ctx.lineWidth = stroke.size;
      ctx.lineCap = 'round';
      ctx.stroke();
    }
  }, [currentStrokeIndex, round]);

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
      <div className="bg-white p-6 rounded shadow-lg max-w-4xl w-full">
        <div className="flex justify-between items-center mb-4">
          <h2 className="text-2xl font-bold">Replay: {round.word}</h2>
          <button onClick={onClose} className="text-gray-500 hover:text-gray-800">Close</button>
        </div>
        
        <canvas
          ref={canvasRef}
          width={800}
          height={600}
          className="bg-white border mb-4"
        />
        
        <div className="flex justify-center gap-4">
          <button onClick={restart} className="bg-gray-200 px-4 py-2 rounded">Restart</button>
          <button onClick={isPlaying ? pause : play} className="bg-primary text-white px-4 py-2 rounded">
            {isPlaying ? 'Pause' : 'Play'}
          </button>
        </div>
      </div>
    </div>
  );
}
```
