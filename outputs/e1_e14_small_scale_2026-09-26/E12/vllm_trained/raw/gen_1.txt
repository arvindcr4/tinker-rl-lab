-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- 1. Profiles Table (Extends Supabase Auth)
create table profiles (
  id uuid references auth.users not null primary key,
  username text,
  avatar_url text,
  updated_at timestamp with time zone default timezone('utc'::text, now())
);

alter table profiles enable row level security;

create policy "Public profiles are viewable by everyone."
  on profiles for select
  using ( true );

create policy "Users can insert their own profile."
  on profiles for insert
  with check ( auth.uid() = id );

create policy "Users can update own profile."
  on profiles for update
  using ( auth.uid() = id );

-- 2. Public Chat Room
create table public_chat (
  id uuid default uuid_generate_v4() primary key,
  user_id uuid references auth.users not null,
  message text not null,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

alter table public_chat enable row level security;

create policy "Anyone can view chat."
  on public_chat for select
  using ( true );

create policy "Authenticated users can post."
  on public_chat for insert
  with check ( auth.uid() is not null );

-- 3. AI Chat History (Private per user)
create table ai_chat_history (
  id uuid default uuid_generate_v4() primary key,
  user_id uuid references auth.users not null,
  role text not null, -- 'user' or 'assistant'
  content text not null,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

alter table ai_chat_history enable row level security;

create policy "Users can view own chat."
  on ai_chat_history for select
  using ( auth.uid() = user_id );

create policy "Users can insert own chat."
  on ai_chat_history for insert
  with check ( auth.uid() = user_id );

-- 4. Realtime Triggers
-- Enable Realtime on tables
alter publication supabase_realtime add table public_chat;
alter publication supabase_realtime add table ai_chat_history;

-- Function to create profile on signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, username)
  values (new.id, split_part(new.email, '@', 1));
  return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

import { NextResponse } from 'next/server';
import { supabase } from '@/lib/supabase';
import { createServerComponentClient } from '@supabase/auth-helpers-nextjs';
import { cookies } from 'next/headers';

export async function GET() {
  const supabase = createServerComponentClient({ cookies });
  const { data, error } = await supabase
    .from('public_chat')
    .select(`
      id, 
      message, 
      created_at, 
      user_id,
      profiles ( username )
    `)
    .order('created_at', { ascending: true })
    .limit(50);

  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json(data);
}

export async function POST(req: Request) {
  const supabase = createServerComponentClient({ cookies });
  const { message } = await req.json();
  
  const { data: { user } } = await supabase.auth.getUser();
  
  if (!user) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  const { data, error } = await supabase
    .from('public_chat')
    .insert({ user_id: user.id, message })
    .select(`
      id, 
      message, 
      created_at, 
      user_id,
      profiles ( username )
    `)
    .single();

  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json(data);
}

  supabase: SupabaseClient;
};

export function SupabaseProvider({ children }: { children: React.ReactNode }) {

    const loadHistory = async () => {
      const { data } = await supabase
        .from('ai_chat_history')
        .select('*')
        .order('created_at', { ascending: true });
      if (data) {
        setMessages(data.map((m: any) => ({ role: m.role, content: m.content })));
      }
    };
    loadHistory();
  }, [supabase]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  // If chart context is provided, add it as a system message or prepend to user message
  const handleSend = async () => {
    if (!input.trim()) return;

    const userMsg: Message = { role: 'user', content: input };
    setMessages(prev => [...prev, userMsg]);
    setInput('');
    setLoading(true);

    // Save user message
    await supabase.from('ai_chat_history').insert({
      user_id: (await supabase.auth.getUser()).data.user?.id,
      role: 'user',
      content: input
    });

    // Prepare context for API
    const contextPayload = chartContext ? {
      start: chartContext.start,
      end: chartContext.end
    } : null;

    try {
      const res = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          messages: [...messages, userMsg],
          context: contextPayload
        })
      });
      const data = await res.json();
      
      const assistantMsg: Message = { role: 'assistant', content: data.reply };
      setMessages(prev => [...prev, assistantMsg]);
      
      // Save assistant message
      await supabase.from('ai_chat_history').insert({
        user_id: (await supabase.auth.getUser()).data.user?.id,
        role: 'assistant',
        content: data.reply
      });
    } catch (error) {
      console.error(error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="h-full flex flex-col border border-terminal-border bg-terminal-panel">
      <div className="p-2 bg-terminal-bg border-b border-terminal-border">
        <h3 className="text-terminal-orange font-bold">AI ASSISTANT</h3>
      </div>
      <div className="flex-1 overflow-y-auto p-2 space-y-2">
        {messages.map((msg, idx) => (
          <div key={idx} className={`p-2 rounded ${
            msg.role === 'user' ? 'bg-terminal-blue/20 text-terminal-blue ml-4' : 'bg-terminal-bg text-terminal-text mr-4'
          }`}>
            <p className="text-sm">{msg.content}</p>
          </div>
        ))}
        {loading && <div className="text-terminal-accent text-sm animate-pulse">ANALYZING...</div>}
        <div ref={messagesEndRef} />
      </div>
      <div className="p-2 border-t border-terminal-border">
        <div className="flex">
          <input
            type="text"
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && handleSend()}
            placeholder="ASK AI..."
            className="flex-1 p-2 bg-terminal-bg border border-terminal-border text-terminal-text focus:outline-none focus:border-terminal-accent"
          />
          <button
            onClick={handleSend}
            disabled={loading}
            className="ml-2 px-4 py-2 bg-terminal-accent text-black font-bold hover:bg-green-400"
          >
            SEND
          </button>
        </div>
        {chartContext && (
          <div className="mt-2 text-xs text-terminal-blue">
            CONTEXT: {new Date(chartContext.start).toLocaleDateString()} - {new Date(chartContext.end).toLocaleDateString()}
          </div>
        )}
      </div>
    </div>
  );
}

'use client';

import { useEffect, useState, useRef } from 'react';
import { useSupabase } from './SupabaseProvider';

interface ChatMessage {
  id: string;
  message: string;
  created_at: string;
  profiles: { username: string };
}

export default function PublicChat() {
  const { supabase } = useSupabase();
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [input, setInput] = useState('');
  const messagesEndRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const fetchMessages = async () => {
      const { data } = await supabase
        .from('public_chat')
        .select(`
          id, 
          message, 
          created_at, 
          profiles ( username )
        `)
        .order('created_at', { ascending: true })
        .limit(50);
      if (data) setMessages(data);
    };

    fetchMessages();

    // Subscribe to realtime changes
    const channel = supabase
      .channel('public_chat_changes')
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'public_chat' }, (payload) => {
        setMessages(prev => [...prev, payload.new as ChatMessage]);
      })
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [supabase]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSend = async () => {
    if (!input.trim()) return;
    await supabase.from('public_chat').insert({ message: input });
    setInput('');
  };

  return (
    <div className="h-full flex flex-col border border-terminal-border bg-terminal-panel">
      <div className="p-2 bg-terminal-bg border-b border-terminal-border">
        <h3 className="text-terminal-blue font-bold">PUBLIC FORUM</h3>
      </div>
      <div className="flex-1 overflow-y-auto p-2 space-y-2">
        {messages.map((msg) => (
          <div key={msg.id} className="p-2 border border-terminal-border bg-terminal-bg">
            <div className="flex justify-between text-xs text-terminal-blue mb-1">
              <span className="font-bold">{msg.profiles?.username || 'Unknown'}</span>
              <span>{new Date(msg.created_at).toLocaleTimeString()}</span>
            </div>
            <p className="text-sm text-terminal-text">{msg.message}</p>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>
      <div className="p-2 border-t border-terminal-border">
        <div className="flex">
          <input
            type="text"
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && handleSend()}
            placeholder="POST TO FORUM..."
            className="flex-1 p-2 bg-terminal-bg border border-terminal-border text-terminal-text focus:outline-none focus:border-terminal-accent"
          />
          <button
            onClick={handleSend}
            className="ml-2 px-4 py-2 bg-terminal-blue text-black font-bold hover:bg-cyan-400"
          >
            POST
          </button>
        </div>
      </div>
    </div>
  );
}

'use client';

import { useEffect, useState } from 'react';
import { useSupabase } from '@/components/SupabaseProvider';
import LoginForm from '@/components/LoginForm';
import Dashboard from '@/components/Dashboard';

export default function Home() {
  const { supabase } = useSupabase();
  const [session, setSession] = useState<any>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      setLoading(false);
    });

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
    });

    return () => subscription.unsubscribe();
  }, [supabase]);

  if (loading) {
    return <div className="flex items-center justify-center h-screen bg-terminal-bg text-terminal-accent">LOADING...</div>;
  }

  if (!session) {
    return <LoginForm />;
  }

  return <Dashboard />;
}
