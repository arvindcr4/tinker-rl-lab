Here is the complete implementation for the AI Legal Assistant.

### Technical Architecture & Decisions
1.  **Framework**: Next.js 14+ (App Router) with TypeScript.
2.  **Database & Auth**: Supabase (PostgreSQL + Auth).
3.  **Vector Database**: ChromaDB (via `chromadb` Python package).
    *   *Note*: Since Next.js runs in a Node.js environment and Chroma is Python-based, this implementation uses a **Hybrid Architecture**. The Next.js app acts as the frontend and API gateway. It communicates with a lightweight Python FastAPI microservice (provided in the `server/` directory) that handles ChromaDB interactions.
4.  **AI Model**: OpenAI (GPT-4o-mini for cost-efficiency and speed, GPT-4o for complex legal reasoning).
5.  **Embeddings**: OpenAI `text-embedding-3-small`.
6.  **Web Search**: SerpAPI (Google Search integration).
7.  **Voice**: Native Web Speech API (Browser-based transcription).
8.  **Styling**: Tailwind CSS.

### Prerequisites
You must provide the following environment variables in your `.env.local` file:
*   `NEXT_PUBLIC_SUPABASE_URL`
*   `NEXT_PUBLIC_SUPABASE_ANON_KEY`
*   `SUPABASE_SERVICE_ROLE_KEY`
*   `OPENAI_API_KEY`
*   `SERPAPI_API_KEY`
*   `CHROMA_HOST` (e.g., `http://localhost:8000` - see setup instructions below)

### 2. ChromaDB Microservice (Python)
This service handles vector storage. It must be running for the RAG features to work.

=== FILE: server/chroma_service.py ===
```python
import chromadb
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional
import uuid

app = FastAPI()

# Initialize ChromaDB client (Persistent client)
# Ensure the directory exists or use in-memory for dev
client = chromadb.PersistentClient(path="./chroma_data")

class DocumentRequest(BaseModel):
    user_id: str
    title: str
    content: str
    metadata: Optional[dict] = {}

class SearchRequest(BaseModel):
    query: str
    user_id: str
    n_results: int = 5
    specific_ids: Optional[List[str]] = []

class DeleteRequest(BaseModel):
    user_id: str
    chroma_id: str

def get_collection(user_id: str):
    """Get or create a collection for a specific user to ensure data isolation."""
    try:
        return client.get_collection(name=f"legal_docs_{user_id}")
    except:
        return client.create_collection(name=f"legal_docs_{user_id}")

@app.post("/add_document")
async def add_document(req: DocumentRequest):
    try:
        collection = get_collection(req.user_id)
        # Generate a unique ID for Chroma
        chroma_id = str(uuid.uuid4())
        
        # ChromaDB requires metadata to be a dict
        metadata = {
            "title": req.title,
            "user_id": req.user_id,
            "source": "uploaded"
        }
        if req.metadata:
            metadata.update(req.metadata)

        collection.add(
            documents=[req.content],
            metadatas=[metadata],
            ids=[chroma_id]
        )
        return {"chroma_id": chroma_id}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/search")
async def search(req: SearchRequest):
    try:
        collection = get_collection(req.user_id)
        
        # If specific IDs are requested (for @mention), we do a direct lookup
        if req.specific_ids:
            results = collection.get(ids=req.specific_ids)
            return results
        else:
            # Vector search
            results = collection.query(
                query_texts=[req.query],
                n_results=req.n_results
            )
            return results
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.delete("/delete_document")
async def delete_document(req: DeleteRequest):
    try:
        collection = get_collection(req.user_id)
        collection.delete(ids=[req.chroma_id])
        return {"success": True}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

export async function generateResponse(
  messages: any[],
  contextDocs: any[],
  webResults: any[]
) {
  // Construct the system prompt with context
  let systemPrompt = `You are an expert AI Legal Assistant. 
  Your goal is to provide accurate, helpful, and legally sound information based on the provided context.
  
  RULES:
  1. Always cite the source of your information. Use the format [Source: Title] or [Source: Web Search Title].
  2. If the provided context does not contain the answer, state that you cannot answer based on the provided documents and suggest a general legal principle if applicable, but clearly distinguish between fact and general knowledge.
  3. Be professional, concise, and objective.
  4. If the user @-mentioned a specific document, prioritize that document's content heavily.
  
  CONTEXT FROM KNOWLEDGE BASE:
  ${contextDocs.map(doc => `Title: ${doc.title}\nContent: ${doc.content}`).join('\n\n---\n\n')}
  
  CONTEXT FROM WEB SEARCH:
  ${webResults.map(res => `Title: ${res.title}\nSnippet: ${res.snippet}`).join('\n\n---\n\n')}
  `;

  const response = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: systemPrompt },
      ...messages
    ],
    temperature: 0.2,
    max_tokens: 1000,
  });

  return response.choices[0].message.content;
}

export async function generateEmbedding(text: string) {
  const response = await openai.embeddings.create({
    model: "text-embedding-3-small",
    input: text,
  });
  return response.data[0].embedding;
}

export async function searchWeb(query: string) {
  const apiKey = process.env.SERPAPI_API_KEY;
  if (!apiKey) return [];

  const url = `https://serpapi.com/search.json?q=${encodeURIComponent(query)}&api_key=${apiKey}&num=5`;
  
  try {
    const res = await fetch(url);
    const data = await res.json();
    
    if (data.organic_results) {
      return data.organic_results.map((r: any) => ({
        title: r.title,
        snippet: r.snippet,
        link: r.link
      }));
    }
  } catch (error) {
    console.error("Web search failed", error);
  }
  return [];
}

const CHROMA_HOST = process.env.CHROMA_HOST || 'http://localhost:8000';

export async function addDocumentToChroma(userId: string, title: string, content: string) {
  const res = await fetch(`${CHROMA_HOST}/add_document`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      user_id: userId,
      title,
      content,
      metadata: { source: 'uploaded' }
    })
  });
  if (!res.ok) throw new Error('Failed to add to Chroma');
  return res.json();
}

export async function searchChroma(userId: string, query: string, nResults: number = 5, specificIds: string[] = []) {
  const res = await fetch(`${CHROMA_HOST}/search`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      user_id: userId,
      query,
      n_results: nResults,
      specific_ids: specificIds
    })
  });
  if (!res.ok) throw new Error('Failed to search Chroma');
  return res.json();
}

export async function deleteDocumentFromChroma(userId: string, chromaId: string) {
  const res = await fetch(`${CHROMA_HOST}/delete_document`, {
    method: 'DELETE',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ user_id: userId, chroma_id: chromaId })
  });
  if (!res.ok) throw new Error('Failed to delete from Chroma');
  return res.json();
}

'use client'

import { useState } from 'react'
import { createClient } from '@/lib/supabase/client'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
import { useRouter } from 'next/navigation'

export default function AuthForm() {
  const [isLogin, setIsLogin] = useState(true)
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [fullName, setFullName] = useState('')
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const router = useRouter()
  const supabase = createClient()

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setLoading(true)
    setError(null)

    try {
      if (isLogin) {
        const { error } = await supabase.auth.signInWithPassword({
          email,
          password,
        })
        if (error) throw error
      } else {
        const { error } = await supabase.auth.signUp({
          email,
          password,
          options: {
            data: {
              full_name: fullName,
            },
          },
        })
        if (error) throw error
        // If signup successful, log them in immediately
        if (!error) {
           const { error: signInError } = await supabase.auth.signInWithPassword({
            email,
            password,
          })
           if (signInError) throw signInError
        }
      }
      router.refresh()
    } catch (error: any) {
      setError(error.message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="flex h-screen w-full items-center justify-center bg-gray-100">
      <Card className="w-full max-w-md">
        <CardHeader>
          <CardTitle>{isLogin ? 'Login' : 'Register'}</CardTitle>
          <CardDescription>
            {isLogin ? 'Enter your credentials to access the legal assistant.' : 'Create an account to get started.'}
          </CardDescription>
        </CardHeader>
        <CardContent>
          <form onSubmit={handleSubmit} className="space-y-4">
            {!isLogin && (
              <div className="space-y-2">
                <Label htmlFor="fullName">Full Name</Label>
                <Input
                  id="fullName"
                  type="text"
                  value={fullName}
                  onChange={(e) => setFullName(e.target.value)}
                  required
                />
              </div>
            )}
            <div className="space-y-2">
              <Label htmlFor="email">Email</Label>
              <Input
                id="email"
                type="email"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                required
              />
            </div>
            <div className="space-y-2">
              <Label htmlFor="password">Password</Label>
              <Input
                id="password"
                type="password"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                required
              />
            </div>
            {error && <p className="text-sm text-red-500">{error}</p>}
            <Button type="submit" className="w-full" disabled={loading}>
              {loading ? 'Processing...' : isLogin ? 'Login' : 'Register'}
            </Button>
          </form>
          <div className="mt-4 text-center text-sm">
            <button
              type="button"

            >
              {isLogin ? "Don't have an account? Register" : "Already have an account? Login"}

  useEffect(() => {
    if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
      const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
      recognitionRef.current = new SpeechRecognition()
      recognitionRef.current.continuous = false
      recognitionRef.current.interimResults = true
      recognitionRef.current.lang = 'en-US'

      recognitionRef.current.onresult = (event: any) => {
        const transcript = Array.from(event.results)
          .map((result: any) => result[0].transcript)
          .join('')
        setTranscript(transcript)
      }

      recognitionRef.current.onend = () => {
        setIsRecording(false)
        // Auto-submit if transcript is substantial
        if (transcript.trim().length > 0) {
          setInput(transcript)
        }
      }
    }
  }, [transcript])

  // Load initial data
  useEffect(() => {
    loadDocuments()
    loadConversation()
  }, [user.id])

  const loadDocuments = async () => {
    const { data, error } = await supabase
      .from('documents')
      .select('*')
      .eq('user_id', user.id)
      .order('created_at', { ascending: false })
    
    if (!error && data) setDocuments(data)
  }

  const loadConversation = async () => {
    // Create a new conversation if none exists
    const newConvId = uuidv4()
    setCurrentConversationId(newConvId)
    
    const { data, error } = await supabase
      .from('chat_messages')
      .select('*')
      .eq('user_id', user.id)
      .eq('conversation_id', newConvId)
      .order('created_at', { ascending: true })

    if (!error && data) setMessages(data)
  }

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
  }

  useEffect(() => {
    scrollToBottom()
  }, [messages])

  const startRecording = () => {
    if (recognitionRef.current) {
      setTranscript('')
      recognitionRef.current.start()
      setIsRecording(true)
    }
  }

  const stopRecording = () => {
    if (recognitionRef.current) {
      recognitionRef.current.stop()
      setIsRecording(false)
    }
  }

  const handleSendMessage = async () => {
    if ((!input.trim() && !transcript.trim()) || isLoading) return

    const userMessageContent = input || transcript
    setInput('')
    setTranscript('')
    
    // Parse @mentions
    const mentionedDocs = documents.filter(doc => 
      userMessageContent.includes(`@${doc.title}`)
    )
    const mentionedIds = mentionedDocs.map(d => d.chroma_id)

    // Add user message to UI
    const newUserMsg: Message = {
      id: uuidv4(),
      role: 'user',
      content: userMessageContent,
      created_at: new Date().toISOString(),
      metadata: { mentionedDocs: mentionedIds }
    }
    setMessages(prev => [...prev, newUserMsg])
    setIsLoading(true)

    try {
      // 1. Save User Message to DB
      await supabase.from('chat_messages').insert({
        user_id: user.id,
        conversation_id: currentConversationId,
        role: 'user',
        content: userMessageContent,
        metadata: { mentionedDocs: mentionedIds }
      })

      // 2. Retrieve Context from Chroma
      let contextDocs = []
      if (mentionedIds.length > 0) {
        // Explicit retrieval for @mentions
        const chromaRes = await searchChroma(user.id, '', 10, mentionedIds)
        if (chromaRes.documents && chromaRes.documents[0]) {
          contextDocs = chromaRes.documents[0].map((doc: string, i: number) => ({
            title: chromaRes.metadatas[0][i].title,
            content: doc
          }))
        }
      } else {
        // Vector search
        const chromaRes = await searchChroma(user.id, userMessageContent, 5)
        if (chromaRes.documents && chromaRes.documents[0]) {
          contextDocs = chromaRes.documents[0].map((doc: string, i: number) => ({
            title: chromaRes.metadatas[0][i].title,
            content: doc
          }))
        }
      }

      // 3. Web Search (if no specific docs mentioned or to supplement)
      let webResults = []
      if (mentionedIds.length === 0) {
        webResults = await searchWeb(userMessageContent)
      }

      // 4. Generate Response
      const aiResponse = await generateResponse(
        messages.map(m => ({ role: m.role, content: m.content })), // Previous context
        contextDocs,
        webResults
      )

      // 5. Save AI Response to UI and DB
      const newAiMsg: Message = {
        id: uuidv4(),
        role: 'assistant',
        content: aiResponse,
        created_at: new Date().toISOString(),
        metadata: { sources: contextDocs.map(d => d.title) }
      }
      setMessages(prev => [...prev, newAiMsg])

      await supabase.from('chat_messages').insert({
        user_id: user.id,
        conversation_id: currentConversationId,
        role: 'assistant',
        content: aiResponse,
        metadata: { sources: contextDocs.map(d => d.title) }
      })

    } catch (error) {
      console.error("Error generating response:", error)
      const errorMsg: Message = {
        id: uuidv4(),
        role: 'assistant',
        content: "I apologize, but I encountered an error processing your request. Please try again.",
        created_at: new Date().toISOString()
      }
      setMessages(prev => [...prev, errorMsg])
    } finally {
      setIsLoading(false)
    }
  }

  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files && e.target.files[0]) {
      const file = e.target.files[0]
      const text = await file.text()
      
      // Add to Chroma
      const chromaRes = await addDocumentToChroma(user.id, uploadTitle, text)
      
      // Save to Supabase
      const { error } = await supabase.from('documents').insert({
        user_id: user.id,
        title: uploadTitle,
        file_name: file.name,
        mime_type: file.type,
        size: file.size,
        chroma_id: chromaRes.chroma_id
      })

      if (!error) {
        setShowUploadModal(false)
        setUploadTitle('')
        setUploadFile(null)
        loadDocuments()
      }
    }
  }

  const handleDeleteDocument = async (doc: Document) => {
    await deleteDocumentFromChroma(user.id, doc.chroma_id)
    await supabase.from('documents').delete().eq('id', doc.id)
    loadDocuments()
  }

  const handleLogout = async () => {
    await supabase.auth.signOut()
    window.location.reload()
  }

  return (
    <div className="flex h-screen w-full overflow-hidden">
      {/* Sidebar */}
      <div className={cn(
        "bg-white border-r border-gray-200 flex flex-col transition-all duration-300 ease-in-out",
        sidebarOpen ? "w-80" : "w-0"
      )}>
        <div className="p-4 border-b border-gray-200 flex justify-between items-center">
          <h2 className="font-semibold text-lg text-gray-800">Knowledge Base</h2>
          <Button variant="ghost" size="icon" onClick={() => setShowUploadModal(true)}>
            <Plus className="h-4 w-4" />
          </Button>
        </div>
        <div className="flex-1 overflow-y-auto p-4 space-y-2">
          {documents.map(doc => (
            <div key={doc.id} className="group flex items-center justify-between p-2 rounded hover:bg-gray-100">
              <div className="flex items-center space-x-2 overflow-hidden">
                <FileText className="h-4 w-4 text-gray-500 flex-shrink-0" />
                <span className="text-sm truncate">{doc.title}</span>
              </div>
              <Button 
                variant="ghost" 
                size="icon" 
                className="h-6 w-6 opacity-0 group-hover:opacity-100"
                onClick={() => handleDeleteDocument(doc)}
              >
                <X className="h-3 w-3" />
              </Button>
            </div>
          ))}
          {documents.length === 0 && (
            <p className="text-sm text-gray-500 text-center mt-4">No documents uploaded.</p>
          )}
        </div>
        <div className="p-4 border-t border-gray-200">
          <Button variant="outline" className="w-full justify-start" onClick={handleLogout}>
            <LogOut className="h-4 w-4 mr-2" />
            Logout
          </Button>
        </div>
      </div>

      {/* Main Chat Area */}
      <div className="flex-1 flex flex-col bg-gray-50">
        {/* Header */}
        <div className="bg-white border-b border-gray-200 p-4 flex justify-between items-center">
          <div className="flex items-center space-x-2">
            <Button variant="ghost" size="icon" onClick={() => setSidebarOpen(!sidebarOpen)}>
              <Menu className="h-5 w-5" />
            </Button>
            <h1 className="font-semibold text-xl text-gray-800">Legal Assistant</h1>
          </div>
          <div className="text-sm text-gray-500">
            {user.email}
          </div>
        </div>

        {/* Messages */}
        <div className="flex-1 overflow-y-auto p-4 space-y-4">
          {messages.map((msg) => (
            <div key={msg.id} className={cn(
              "flex w-full",
              msg.role === 'user' ? "justify-end" : "justify-start"
            )}>
              <div className={cn(
                "max-w-[80%] rounded-lg p-4",
                msg.role === 'user' 
                  ? "bg-blue-600 text-white" 
                  : "bg-white border border-gray-200 text-gray-800"
              )}>
                <div className="prose prose-sm max-w-none">
                  <ReactMarkdown remarkPlugins={[remarkGfm]}>
                    {msg.content}
                  </ReactMarkdown>
                </div>
                {msg.metadata?.sources && msg.role === 'assistant' && (
                  <div className="mt-2 text-xs opacity-70">
                    Sources: {msg.metadata.sources.join(', ')}
                  </div>
                )}
              </div>
            </div>
          ))}
          {isLoading && (
            <div className="flex justify-start">
              <div className="bg-white border border-gray-200 rounded-lg p-4 flex items-center space-x-2">
                <Loader2 className="h-4 w-4 animate-spin text-blue-600" />
                <span className="text-sm text-gray-500">Thinking...</span>
              </div>
            </div>
          )}
          <div ref={messagesEndRef} />
        </div>

        {/* Input Area */}
        <div className="bg-white border-t border-gray-200 p-4">
          <div className="max-w-4xl mx-auto relative">
            <Textarea
              value={input}
              onChange={(e) => setInput(e.target.value)}
              placeholder="Ask a legal question... (Use @DocumentName to reference specific files)"
              className="pr-24 min-h-[60px] resize-none"
              onKeyDown={(e) => {
                if (e.key === 'Enter' && !e.shiftKey) {
                  e.preventDefault()
                  handleSendMessage()
                }
              }}
            />
            <div className="absolute right-2 bottom-2 flex items-center space-x-2">
              <Button
                variant="ghost"
                size="icon"
                className={cn(
                  "h-8 w-8 rounded-full",
                  isRecording ? "bg-red-100 text-red-600" : "text-gray-500"
                )}
                onClick={isRecording ? stopRecording : startRecording}
              >
                {isRecording ? <MicOff className="h-4 w-4" /> : <Mic className="h-4 w-4" />}
              </Button>
              <Button 
                size="icon" 
                className="h-8 w-8 rounded-full bg-blue-600 hover:bg-blue-700"
                onClick={handleSendMessage}
                disabled={isLoading}
              >
                <Send className="h-4 w-4" />
              </Button>
            </div>
          </div>
          {isRecording && (
            <div className="text-center mt-2 text-sm text-red-500 animate-pulse">
              Listening... Speak now.
            </div>
          )}
        </div>
      </div>

      {/* Upload Modal */}
      {showUploadModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
          <Card className="w-full max-w-md">
            <CardHeader>
              <CardTitle>Upload Document</CardTitle>
            </CardHeader>
            <CardContent className="space-y-4">
              <div>
                <Label>Title</Label>
                <Input 
                  value={uploadTitle} 
                  onChange={(e) => setUploadTitle(e.target.value)} 
                  placeholder="e.g., Employment Contract"
                />
              </div>
              <div>
                <Label>File (Text supported)</Label>
                <Input 
                  type="file" 
                  onChange={(e) => setUploadFile(e.target.files?.[0] || null)} 
                  accept=".txt,.md,.docx"
                />
              </div>
              <div className="flex justify-end space-x-2">
                <Button variant="outline" onClick={() => setShowUploadModal(false)}>Cancel</Button>
                <Button 
                  onClick={handleFileUpload} 
                  disabled={!uploadTitle || !uploadFile}
                >
                  Upload
                </Button>
              </div>
            </CardContent>
          </Card>
        </div>
      )}
    </div>
  )
}

### Setup Instructions

1.  **Initialize Project**:
    ```bash
    npx create-next-app@latest legal-ai-assistant --typescript --tailwind --app
    cd legal-ai-assistant
    npm install @supabase/ssr @supabase/supabase-js lucide-react react-markdown remark-gfm date-fns clsx tailwind-merge uuid
    npm install -D @types/uuid
    ```

2.  **Install ChromaDB Microservice Dependencies**:
    ```bash
    mkdir server
    cd server
    pip install -r requirements.txt
    ```

3.  **Environment Variables**:
    Create `.env.local` in the root directory with your keys.

4.  **Run ChromaDB Service**:
    ```bash
    cd server
    python chroma_service.py
    ```

5.  **Run Next.js App**:
    ```bash
    npm run dev
    ```

6.  **Database Setup**:
    Copy the content of `supabase_schema.sql` into your Supabase SQL Editor and run it.

7.  **Usage**:
    *   Register/Login.
    *   Upload a text document (e.g., a contract) via the sidebar.
    *   Ask a question in the chat.
    *   Use `@Contract_2024` to reference a specific document.
    *   Use the microphone icon to dictate questions.
