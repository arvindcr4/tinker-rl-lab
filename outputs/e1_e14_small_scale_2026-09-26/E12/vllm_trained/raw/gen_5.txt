Here is the complete implementation for the multiplayer vocabulary-drawing game.

### Technical Architecture
*   **Framework**: Next.js 14+ (App Router).
*   **Language**: TypeScript.
*   **Backend/Database**: Supabase (PostgreSQL, Auth, Realtime).
*   **Real-time Communication**: Supabase Realtime (for game state, chat, and lobby updates) + WebSockets (via Supabase).
*   **Drawing Engine**: HTML5 Canvas with a custom stroke history system for replay.
*   **Styling**: Tailwind CSS.

### Prerequisites
You must create a Supabase project and run the SQL provided in `supabase_schema.sql` in your Supabase SQL Editor. You must also set the environment variables in your `.env.local` file.

```sql
-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- 1. Users Profile (extends Supabase Auth)
create table profiles (
  id uuid references auth.users not null primary key,
  username text unique not null,
  score int default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Game Rooms
create table game_rooms (
  id uuid default uuid_generate_v4() primary key,
  name text not null,
  host_id uuid references auth.users(id),
  status text default 'waiting' check (status in ('waiting', 'playing', 'finished')),
  max_players int default 6,
  current_players int default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Room Players
create table room_players (
  id uuid default uuid_generate_v4() primary key,
  room_id uuid references game_rooms(id) on delete cascade,
  user_id uuid references auth.users(id),
  score int default 0,
  is_drawing boolean default false,
  joined_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 4. Game Rounds (One round per word)
create table game_rounds (
  id uuid default uuid_generate_v4() primary key,
  room_id uuid references game_rooms(id) on delete cascade,
  word text not null,
  drawing_user_id uuid references auth.users(id),
  status text default 'active' check (status in ('active', 'finished')),
  started_at timestamp with time zone default timezone('utc'::text, now()) not null,
  finished_at timestamp with time zone
);

-- 5. Strokes (Drawing Data)
create table strokes (
  id uuid default uuid_generate_v4() primary key,
  round_id uuid references game_rounds(id) on delete cascade,
  user_id uuid references auth.users(id),
  data jsonb not null, -- { x, y, color, size, type }
  timestamp bigint not null -- for replay timing
);

-- 6. Chat/Guesses
create table game_messages (
  id uuid default uuid_generate_v4() primary key,
  room_id uuid references game_rooms(id) on delete cascade,
  user_id uuid references auth.users(id),
  message text not null,
  is_guess boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Row Level Security (RLS) Policies
alter table profiles enable row level security;
alter table game_rooms enable row level security;
alter table room_players enable row level security;
alter table game_rounds enable row level security;
alter table strokes enable row level security;
alter table game_messages enable row level security;

-- Profiles: Users can read own profile
create policy "Public profiles are viewable by everyone."
  on profiles for select
  using ( true );

create policy "Users can insert own profile."
  on profiles for insert
  with check ( auth.uid() = id );

create policy "Users can update own profile."
  on profiles for update
  using ( auth.uid() = id );

-- Game Rooms: Anyone can view, only host can update status
create policy "Rooms are viewable by everyone."
  on game_rooms for select
  using ( true );

create policy "Host can update room."
  on game_rooms for update
  using ( auth.uid() = host_id );

-- Room Players: Viewable by room members
create policy "Room players viewable by room members."
  on room_players for select
  using ( 
    exists (
      select 1 from game_rooms 
      where game_rooms.id = room_players.room_id 
      and game_rooms.status != 'finished'
    )
  );

-- Strokes: Viewable by room members
create policy "Strokes viewable by room members."
  on strokes for select
  using ( 
    exists (
      select 1 from game_rounds gr
      join game_rooms grm on grm.id = gr.room_id
      where grm.id = (select room_id from room_players where user_id = auth.uid())
    )
  );

-- Messages: Viewable by room members
create policy "Messages viewable by room members."
  on game_messages for select
  using ( 
    exists (
      select 1 from game_rooms grm
      where grm.id = game_messages.room_id
      and grm.status != 'finished'
    )
  );

-- Trigger to create profile on signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, username, score)
  values (new.id, new.email, 0);
  return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();
```

```tsx
    const y = e.clientY - rect.top;

    ctx.lineTo(x, y);
    ctx.stroke();

    setCurrentStroke(prev => [...prev, { x, y, color: tool === 'eraser' ? '#ffffff' : color, size: brushSize, type: 'line' }]);

      // For now, we assume the backend handles the round ID association via a trigger or the client knows it.
      // Let's assume we have a way to get the current round ID.

      // await supabase.from('strokes').insert({ round_id: currentRoundId, data: currentStroke, ... });

      setCurrentStroke([]);
    }
  };

    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;
    ctx.clearRect(0, 0, canvas.width, canvas.height);

  };

  return (

      <div className="mb-4 flex gap-2">

        <input

        <button 
          onClick={() => setTool(tool === 'brush' ? 'eraser' : 'brush')}

        >

          className="px-4 py-2 bg-red-500 text-white rounded"
```

```tsx
'use client';

import { useEffect, useRef, useState } from 'react';

interface Stroke {
  x: number;
  y: number;
  color: string;
  size: number;
  type: 'move' | 'line';
}

interface ReplayProps {
  strokes: Stroke[][];
  isPlaying: boolean;
  onPlay: () => void;
  onPause: () => void;
  onRestart: () => void;
}

export default function Replay({ strokes, isPlaying, onPlay, onPause, onRestart }: ReplayProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [currentStrokeIndex, setCurrentStrokeIndex] = useState(0);
  const [currentPointIndex, setCurrentPointIndex] = useState(0);
  const [isFinished, setIsFinished] = useState(false);

  useEffect(() => {
    if (!isPlaying || isFinished) return;

    const interval = setInterval(() => {
      if (currentStrokeIndex >= strokes.length) {
        setIsFinished(true);
        onPause();
        return;
      }

      const currentStroke = strokes[currentStrokeIndex];
      if (currentPointIndex < currentStroke.length) {
        drawPoint(currentStroke[currentPointIndex]);
        setCurrentPointIndex(prev => prev + 1);
      } else {
        setCurrentStrokeIndex(prev => prev + 1);
        setCurrentPointIndex(0);
      }
    }, 50); // Speed of replay

    return () => clearInterval(interval);
  }, [isPlaying, currentStrokeIndex, currentPointIndex, strokes, isFinished, onPause]);

  const drawPoint = (point: Stroke) => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    if (point.type === 'move') {
      ctx.beginPath();
      ctx.moveTo(point.x, point.y);
    } else {
      ctx.lineTo(point.x, point.y);
      ctx.stroke();
    }
    ctx.strokeStyle = point.color;
    ctx.lineWidth = point.size;
    ctx.lineCap = 'round';
  };

  const renderAll = () => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    strokes.forEach(stroke => {
      stroke.forEach((point, i) => {
        if (i === 0) {
          ctx.beginPath();
          ctx.moveTo(point.x, point.y);
        } else {
          ctx.lineTo(point.x, point.y);
        }
      });
      ctx.strokeStyle = stroke[0].color;
      ctx.lineWidth = stroke[0].size;
      ctx.lineCap = 'round';
      ctx.stroke();
    });
  };

  useEffect(() => {
    renderAll();
  }, [strokes]);

  return (
    <div className="flex flex-col items-center">
      <canvas
        ref={canvasRef}
        width={800}
        height={600}
        className="border-2 border-gray-300 bg-white"
      />
      <div className="mt-4 flex gap-4">
        {!isFinished && (
          <button onClick={isPlaying ? onPause : onPlay} className="px-4 py-2 bg-primary text-white rounded">
            {isPlaying ? 'Pause' : 'Play'}
          </button>
        )}
        <button onClick={onRestart} className="px-4 py-2 bg-gray-500 text-white rounded">
          Restart
        </button>
      </div>
    </div>
  );
}
```

```tsx
'use client';

import { useEffect, useState, useRef } from 'react';
import { supabase } from '@/lib/supabase';
import { useRouter } from 'next/navigation';
import Canvas from '@/components/Canvas';
import Replay from '@/components/Replay';
import { getRandomWord } from '@/lib/utils';

interface Player {
  user_id: string;
  username: string;
  score: number;
  is_drawing: boolean;
}

interface Message {
  user_id: string;
  username: string;
  message: string;
  is_guess: boolean;
}

export default function RoomPage({ params }: { params: { roomId: string } }) {
  const router = useRouter();
  const [players, setPlayers] = useState<Player[]>([]);
  const [messages, setMessages] = useState<Message[]>([]);
  const [currentWord, setCurrentWord] = useState<string>('');
  const [isDrawing, setIsDrawing] = useState(false);
  const [myUserId, setMyUserId] = useState<string>('');
  const [roundStatus, setRoundStatus] = useState<'active' | 'finished'>('active');
  const [strokes, setStrokes] = useState<any[][]>([]);
  const [isReplaying, setIsReplaying] = useState(false);
  const [guessInput, setGuessInput] = useState('');
  const [usedWords, setUsedWords] = useState<string[]>([]);

  const messagesEndRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const init = async () => {
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) {
        router.push('/');
        return;
      }
      setMyUserId(user.id);

      // Fetch initial room data
      const { data: roomPlayers } = await supabase
        .from('room_players')
        .select(`
          user_id,
          score,
          is_drawing,
          profiles(username)
        `)
        .eq('room_id', params.roomId);

      if (roomPlayers) {
        setPlayers(roomPlayers.map(p => ({
          user_id: p.user_id,
          username: p.profiles?.username || 'Unknown',
          score: p.score,
          is_drawing: p.is_drawing
        })));
      }

      // Check if I am the host
      const { data: room } = await supabase
        .from('game_rooms')
        .select('host_id')
        .eq('id', params.roomId)
        .single();

      if (room?.host_id === user.id) {
        startRound();
      }

      // Listen for player updates
      const channel = supabase
        .channel(`room:${params.roomId}`)
        .on('postgres_changes', { event: '*', schema: 'public', table: 'room_players' }, (payload) => {
          if (payload.eventType === 'INSERT') {
            // New player joined
            fetchPlayers();
          } else if (payload.eventType === 'DELETE') {
            // Player left
            fetchPlayers();
          } else if (payload.eventType === 'UPDATE') {
            // Score or drawing status changed
            fetchPlayers();
          }
        })
        .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'game_messages' }, (payload) => {
          const msg = payload.new as any;
          setMessages(prev => [...prev, {
            user_id: msg.user_id,
            username: msg.profiles?.username || 'Unknown',
            message: msg.message,
            is_guess: msg.is_guess
          }]);
        })
        .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'strokes' }, (payload) => {
          const stroke = payload.new.data;
          setStrokes(prev => [...prev, [stroke]]);
        })
        .subscribe();

      return () => {
        supabase.removeChannel(channel);
      };
    };

    init();
  }, [params.roomId, router]);

  const fetchPlayers = async () => {
    const { data } = await supabase
      .from('room_players')
      .select(`
        user_id,
        score,
        is_drawing,
        profiles(username)
      `)
      .eq('room_id', params.roomId);

    if (data) {
      setPlayers(data.map(p => ({
        user_id: p.user_id,
        username: p.profiles?.username || 'Unknown',
        score: p.score,
        is_drawing: p.is_drawing
      })));
    }
  };

  const startRound = async () => {
    const word = getRandomWord(usedWords);
    setUsedWords(prev => [...prev, word]);
    setCurrentWord(word);
    setRoundStatus('active');
    setIsReplaying(false);
    setStrokes([]);

    // Update drawing player
    const drawingPlayer = players.find(p => p.is_drawing);
    if (!drawingPlayer) {
      // Assign first player as drawer
      const firstPlayer = players[0];
      if (firstPlayer) {
        await supabase
          .from('room_players')
          .update({ is_drawing: true })
          .eq('room_id', params.roomId);
        setIsDrawing(firstPlayer.user_id === myUserId);
      }
    } else {
      setIsDrawing(drawingPlayer.user_id === myUserId);
    }
  };

  const endRound = async () => {
    setRoundStatus('finished');
    setIsReplaying(true);
    
    // Reset drawing status
    await supabase
      .from('room_players')
      .update({ is_drawing: false })
      .eq('room_id', params.roomId);

    // Update scores
    // In a real app, we'd calculate scores based on guesses.
    // For now, we just increment the drawer's score for completing the round.
    const drawer = players.find(p => p.is_drawing);
    if (drawer) {
      await supabase.rpc('increment_score', { player_id: drawer.user_id, amount: 100 });
    }
  };

  const handleGuess = async () => {
    if (!guessInput.trim()) return;

    const { data: { user } } = await supabase.auth.getUser();
    if (!user) return;

    await supabase.from('game_messages').insert({
      room_id: params.roomId,
      user_id: user.id,
      message: guessInput,
      is_guess: true
    });

    // Check if guess is correct
    if (guessInput.toLowerCase().trim() === currentWord.toLowerCase().trim()) {
      // Correct guess!
      await supabase.rpc('increment_score', { player_id: user.id, amount: 100 });
      // Notify others?
      await supabase.from('game_messages').insert({
        room_id: params.roomId,
        user_id: user.id,
        message: `Correct! The word was ${currentWord}`,
        is_guess: false
      });
      endRound();
    }

    setGuessInput('');
  };

  const handleReplayControl = (action: 'play' | 'pause' | 'restart') => {
    if (action === 'play') setIsReplaying(true);
    if (action === 'pause') setIsReplaying(false);
    if (action === 'restart') {
      setIsReplaying(false);
      // Reset canvas state for replay
    }
  };

  return (
    <div className="min-h-screen flex flex-col">
      <header className="bg-white shadow p-4 flex justify-between items-center">
        <h1 className="text-xl font-bold text-primary">Room: {params.roomId.slice(0, 8)}</h1>
        <button onClick={() => router.push('/')} className="text-gray-600">Leave</button>
      </header>

      <div className="flex-1 flex overflow-hidden">
        {/* Sidebar: Players & Chat */}
        <div className="w-1/3 bg-white border-r p-4 flex flex-col">
          <div className="mb-4">
            <h2 className="font-bold mb-2">Players</h2>
            <ul className="space-y-2">
              {players.map(player => (
                <li key={player.user_id} className="flex justify-between">
                  <span>{player.username} {player.user_id === myUserId && '(You)'}</span>
                  <span>{player.score} pts</span>
                </li>
              ))}
            </ul>
          </div>

          <div className="flex-1 overflow-y-auto mb-4">
            <h2 className="font-bold mb-2">Chat</h2>
            <div className="space-y-2">
              {messages.map((msg, i) => (
                <div key={i} className={`text-sm ${msg.is_guess ? 'text-blue-600' : 'text-gray-600'}`}>
                  <strong>{msg.username}:</strong> {msg.message}
                </div>
              ))}
            </div>
            <div ref={messagesEndRef} />
          </div>

          <div className="flex gap-2">
            <input
              type="text"
              value={guessInput}
              onChange={(e) => setGuessInput(e.target.value)}
              placeholder="Type your guess..."
              className="flex-1 p-2 border rounded"
              onKeyDown={(e) => e.key === 'Enter' && handleGuess()}
            />
            <button onClick={handleGuess} className="bg-primary text-white px-4 rounded">
              Guess
            </button>
          </div>
        </div>

        {/* Main: Canvas or Replay */}
        <div className="flex-1 bg-gray-100 p-4 flex items-center justify-center">
          {roundStatus === 'active' ? (
            <Canvas
              roomId={params.roomId}
              isDrawing={isDrawing}
              word={isDrawing ? currentWord : undefined}
              onRoundEnd={endRound}
            />
          ) : (
            <Replay
              strokes={strokes}
              isPlaying={isReplaying}
              onPlay={() => handleReplayControl('play')}
              onPause={() => handleReplayControl('pause')}
              onRestart={() => handleReplayControl('restart')}
            />
          )}
        </div>
      </div>
    </div>
  );
}
```
