+/* Helper function to safely convert a Lua number to a long long.
+ * Returns C_OK on success and sets *val to the converted value.
+ * Returns C_ERR if the value is out of range or not an integer. */
+static int luaNumberToLongLong(lua_State *L, long long *val) {
+    if (!lua_isnumber(L, -1)) {
+        return C_ERR;
+    }
+    lua_Number n = lua_tonumber(L, -1);
+    /* Check for NaN or Infinity */
+    if (n != n || n == HUGE_VAL || n == -HUGE_VAL) {
+        return C_ERR;
+    }
+    /* Check for integer range */
+    if (n < (lua_Number)LONG_MIN || n > (lua_Number)LONG_MAX) {
+        return C_ERR;
+    }
+    /* Check for fractional part */
+    if (n != (lua_Number)(long long)n) {
+        return C_ERR;
+    }
+    *val = (long long)n;
+    return C_OK;
+}
