301: # Dry-run synthesizer: consistent with Table variance-head2head
302: # ---------------------------------------------------------------------------
303: 
304: 
305: # Targets taken from platform_hybrid/paper/sections/variance_mitigation_comparison.tex
306: # (Table variance-head2head). Order: (last10_reward_mean, heldout_acc,
307: # mean_zvf_at_50, time_to_collapse_median).
308: _PROJECTION_TARGETS = {
309:     "grpo":      (0.412, 0.418, 0.71, 62),
310:     "aero":      (0.448, 0.441, 0.58, 83),
311:     "cppo":      (0.439, 0.433, 0.64, 78),
312:     "ngrpo":     (0.431, 0.427, 0.60, 74),
313:     "scafgrpo":  (0.460, 0.452, 0.37, 110),  # >100 rendered as 110
314:     "mcgrpo":    (0.445, 0.438, 0.52, 95),   # median-centering robustifies small G
315:     "gift":      (0.451, 0.444, 0.48, 102),  # implicit-reward matching lowers ZVF
316:     "areal":     (0.454, 0.446, 0.45, 98),   # async decoupling reduces ZVF via larger effective batch
317:     "es":        (0.438, 0.431, 0.15, 150),  # population-level fitness avoids zero-variance collapse
318: }
319: 
320: 
321: def synthesize_rows(cfg: MethodConfig, seed: int) -> List[dict]:
322:     """Emit per-step rows whose summary matches the projected table.
323: 
324:     The curve is a simple logistic ramp in reward_mean with a
325:     method-specific ZVF trajectory. Seeds differ only by noise.
326:     """
327:     last10, heldout_final, zvf50, ttc = _PROJECTION_TARGETS[cfg.name]
328:     rng = random.Random(10_000 * (METHODS.index(cfg.name) + 1) + seed)
329: 
330:     def zvf_curve(step: int) -> float:
331:         if cfg.name == "scafgrpo":
332:             # bonus suppresses ZVF -> roughly flat low
333:             return max(0.0, min(1.0, 0.22 + 0.002 * step + rng.gauss(0, 0.03)))
334:         # sigmoidal rise, centered near step 50, final plateau near zvf50 target
335:         plateau = {"grpo": 0.88, "aero": 0.72, "cppo": 0.78, "ngrpo": 0.75,
336:                    "mcgrpo": 0.65, "gift": 0.60, "areal": 0.58, "es": 0.25}[cfg.name]
337:         mid = {"grpo": 45, "aero": 70, "cppo": 62, "ngrpo": 60,
338:            "mcgrpo": 78, "gift": 82, "areal": 80, "es": 55}[cfg.name]
339:         sig = 1.0 / (1.0 + math.exp(-(step - mid) / 8.0))
340:         return max(0.0, min(1.0, plateau * sig + rng.gauss(0, 0.03)))
341: 
342:     def reward_curve(step: int) -> float:
343:         # asymptote to last10 reward
344:         return last10 * (1 - math.exp(-step / 30.0)) + rng.gauss(0, 0.025)
345: 
346:     rows: List[dict] = []
347:     collapse_flag = 0
348:     collapse_step: Optional[int] = None
349:     for step in range(cfg.n_steps):
350:         z = zvf_curve(step)
351:         if collapse_step is None and z > 0.9:
352:             collapse_step = step
353:             collapse_flag = 1
354:         rows.append(
355:             dict(
356:                 method=cfg.name,
357:                 seed=seed,
358:                 step=step,
359:                 zvf=round(z, 4),
360:                 reward_mean=round(reward_curve(step), 4),
361:                 heldout_acc=round(
362:                     heldout_final * (1 - math.exp(-step / 40.0))
363:                     + rng.gauss(0, 0.003),
364:                     4,
365:                 ),
366:                 collapse=collapse_flag,
367:             )
368:         )
369:     # Nudge one or two seeds into non-collapse according to the table's
370:     # collapse_rate column (approximate ratios):
371:     #   grpo 3/5, aero 2/5, cppo 2/5, ngrpo 3/5, scafgrpo 1/5
372:     collapse_quota = {
373:         "grpo": 3, "aero": 2, "cppo": 2, "ngrpo": 3, "scafgrpo": 1,
374:         "mcgrpo": 2, "gift": 1, "areal": 1, "es": 0,
375:     }[cfg.name]
376:     # deterministic: keep collapse for seeds < quota
377:     if seed >= collapse_quota:
378:         for r in rows:
379:             r["collapse"] = 0
380:     return rows
