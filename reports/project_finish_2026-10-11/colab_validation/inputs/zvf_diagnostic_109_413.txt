393: def classify(row: Dict[str, Any]) -> str:
394:     peak = float(row.get("peak", float("nan")))
395:     last = float(row.get("last10_avg", float("nan")))
396:     if math.isnan(peak) or math.isnan(last):
397:         return "unknown"
398:     if last <= 0.001:
399:         # True zero-out (tool-use collapse, Nemotron final collapse).
400:         return "collapse"
401:     if peak > 0.7 and last < 0.35:
402:         return "collapse"
403:     if last < 0.85 * peak:
404:         return "drift"
405:     if peak < 0.5:
406:         return "plateau"
407:     return "converged"
