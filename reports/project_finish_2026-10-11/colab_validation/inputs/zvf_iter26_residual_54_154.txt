109: def analyse_slice(slice_rows, label, B=2000, seed=0):
110:     """Run naive + residual + reward correlations on one slice."""
111:     n = len(slice_rows)
112:     out = {"label": label, "n_pooled": n}
113:     if n < 5:
114:         out.update({"naive_r": "NA", "reward_r": "NA", "residual_r": "NA",
115:                     "naive_lo": "NA", "naive_hi": "NA",
116:                     "residual_lo": "NA", "residual_hi": "NA"})
117:         return out
118:     zvfs = [r["mean_zvf"] for r in slice_rows]
119:     lasts = [r["last10"] for r in slice_rows]
120:     rewards = [r["mean_reward"] for r in slice_rows]
121: 
122:     naive = _bootstrap(zvfs, lasts, _pearson, B=B, seed=seed)
123:     reward = _bootstrap(rewards, lasts, _pearson, B=B, seed=seed + 1)
124:     # Residual last10 ~ reward
125:     mx = statistics.fmean(rewards)
126:     my = statistics.fmean(lasts)
127:     num = sum((x - mx) * (y - my) for x, y in zip(rewards, lasts))
128:     den = sum((x - mx) ** 2 for x in rewards)
129:     b1 = num / den if den != 0 else 0.0
130:     b0 = my - b1 * mx
131:     resid = [y - (b0 + b1 * x) for x, y in zip(rewards, lasts)]
132:     residual = _bootstrap(zvfs, resid, _pearson, B=B, seed=seed + 2)
133: 
134:     out.update(
135:         {
136:             "naive_r": f"{naive[0]:+.3f}",
137:             "naive_lo": f"{naive[1][0]:+.3f}",
138:             "naive_hi": f"{naive[1][1]:+.3f}",
139:             "reward_r": f"{reward[0]:+.3f}",
140:             "residual_r": f"{residual[0]:+.3f}",
141:             "residual_lo": f"{residual[1][0]:+.3f}",
142:             "residual_hi": f"{residual[1][1]:+.3f}",
143:         }
144:     )
145:     return out
